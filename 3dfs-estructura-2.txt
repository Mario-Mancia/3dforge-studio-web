#!/usr/bin/env python
"""Django's command-line utility for administrative tasks."""
import os
import sys


def main():
    """Run administrative tasks."""
    os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'config.settings')
    try:
        from django.core.management import execute_from_command_line
    except ImportError as exc:
        raise ImportError(
            "Couldn't import Django. Are you sure it's installed and "
            "available on your PYTHONPATH environment variable? Did you "
            "forget to activate a virtual environment?"
        ) from exc
    execute_from_command_line(sys.argv)


if __name__ == '__main__':
    main()

from decimal import ROUND_HALF_UP, Decimal, InvalidOperation

from apps.core.constants import CENTAVO


def redondear(valor):
    return Decimal(valor).quantize(CENTAVO, rounding=ROUND_HALF_UP)


def importes_coherentes(subtotal, descuento, impuesto, total):
    try:
        esperado = redondear(subtotal) - redondear(descuento) + redondear(impuesto)
        return esperado == redondear(total)
    except (InvalidOperation, TypeError, ValueError):
        return True

cantidad_minima = MinValueValidator(Decimal("0.01"))


def validar_ruta_relativa(valor):
    texto = str(valor)
    partes = re.split(r"[\\/]", texto)
    if (
        not texto
        or len(texto) > 500
        or texto != texto.strip()
        or texto.startswith(("/", "\\", "~"))
        or re.match(r"^[A-Za-z]:", texto)
        or "://" in texto
        or any(parte in ("", ".", "..") for parte in partes)
        or any(ord(caracter) < 32 for caracter in texto)
    ):
        raise ValidationError("Ruta de archivo no permitida.", code="ruta_invalida")


def validar_nombre_archivo(valor):
    texto = str(valor)
    if (
        not texto
        or texto != texto.strip()
        or texto.startswith(".")
        or re.search(r"[\\/\x00-\x1f]", texto)
    ):
        raise ValidationError("Nombre de archivo no permitido.", code="archivo_invalido")


def validar_tamano_archivo(valor):
    maximo = getattr(settings, "MAX_ARCHIVO_BYTES", MAX_ARCHIVO_BYTES)
    if valor > maximo:
        raise ValidationError(f"El archivo supera el maximo permitido de {maximo} bytes.", code="archivo_grande")


def validar_objeto_json(valor):
    if valor is None:
        return
    if not isinstance(valor, dict):
        raise ValidationError("Debe ser un objeto JSON.", code="json_invalido")
    try:
        longitud = len(json.dumps(valor, ensure_ascii=False))
    except (TypeError, ValueError):
        raise ValidationError("El objeto JSON no es serializable.", code="json_invalido")
    if longitud > MAX_JSON_BYTES:
        raise ValidationError("El objeto JSON es demasiado grande.", code="json_grande")

from apps.usuarios.managers import UsuarioManager


class Usuario(ModeloValidado, AbstractBaseUser, PermissionsMixin):
    nombres = models.CharField(max_length=100, validators=[validar_nombre_persona])
    apellidos = models.CharField(max_length=100, validators=[validar_nombre_persona])
    correo = models.EmailField(max_length=150, unique=True)
    telefono = models.CharField(max_length=25, null=True, blank=True, validators=[validar_telefono])
    is_staff = models.BooleanField(default=False)
    is_active = models.BooleanField(default=True)
    date_joined = models.DateTimeField(default=timezone.now)
    estado = models.ForeignKey(
        "usuarios.EstadoUsuario",
        on_delete=models.PROTECT,
        db_column="id_estado",
        related_name="usuarios",
        default=ESTADO_ACTIVO,
    )

    objects = UsuarioManager()

    USERNAME_FIELD = "correo"
    EMAIL_FIELD = "correo"
    REQUIRED_FIELDS = ["nombres", "apellidos"]
    campos_texto_opcional = ("telefono",)

    class Meta:
        db_table = "usuarios"
        verbose_name = "usuario"
        verbose_name_plural = "usuarios"

    def __str__(self):
        return self.correo

    @property
    def nombre_completo(self):
        return f"{self.nombres} {self.apellidos}".strip()

    def get_full_name(self):
        return self.nombre_completo

    def get_short_name(self):
        return self.nombres

    def clean(self):
        super().clean()
        if self.correo:
            self.correo = self.correo.strip().lower()
        if self.estado_id in ESTADOS_BLOQUEADOS:
            self.is_active = False

    def tiene_algun_rol(self, roles):
        if not (self.is_active and self.estado_id == ESTADO_ACTIVO):
            return False
        if self.is_superuser:
            return True
        return self.groups.filter(name__in=roles).exists()

from django import forms
from django.core.exceptions import ValidationError

from apps.catalogo.models import CategoriaServicio, Producto, Servicio
from apps.catalogo.services.catalogo import eliminar_imagen, guardar_imagen
from apps.core.constants import FORMATOS_IMAGEN, MAX_IMAGEN_BYTES

from django.conf import settings

FORMATOS_PILLOW = {"JPEG", "PNG", "WEBP"}


class ProductoForm(forms.ModelForm):
    imagen_archivo = forms.ImageField(required=False, label="Imagen")
    quitar_imagen = forms.BooleanField(required=False, label="Quitar imagen actual")

    class Meta:
        model = Producto
        fields = ["sku", "nombre", "descripcion", "precio", "stock", "stock_minimo", "activo"]
        widgets = {"descripcion": forms.Textarea(attrs={"rows": 4})}

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        if self.instance.pk:
            self.fields.pop("stock")

    def clean_imagen_archivo(self):
        archivo = self.cleaned_data.get("imagen_archivo")
        if not archivo:
            return archivo
        extension = archivo.name.rsplit(".", 1)[-1].lower() if "." in archivo.name else ""
        if extension not in FORMATOS_IMAGEN:
            raise ValidationError("Formato no permitido. Use JPG, PNG o WEBP.")
        maximo = getattr(settings, "IMAGEN_MAX_BYTES", MAX_IMAGEN_BYTES)
        if archivo.size > maximo:
            raise ValidationError("La imagen supera el tamaño máximo permitido.")
        imagen = getattr(archivo, "image", None)
        if imagen is None or imagen.format not in FORMATOS_PILLOW:
            raise ValidationError("El archivo no es una imagen válida.")
        return archivo

    def save(self, commit=True):
        producto = super().save(commit=False)
        archivo = self.cleaned_data.get("imagen_archivo")
        anterior = producto.imagen
        if archivo:
            producto.imagen = guardar_imagen(archivo)
        elif self.cleaned_data.get("quitar_imagen"):
            producto.imagen = None
        if commit:
            producto.save()
            if anterior and anterior != producto.imagen:
                eliminar_imagen(anterior)
        return producto


class AjusteStockForm(forms.Form):
    delta = forms.IntegerField(min_value=-100000, max_value=100000, label="Ajuste")

    def clean_delta(self):
        delta = self.cleaned_data["delta"]
        if delta == 0:
            raise ValidationError("El ajuste no puede ser cero.")
        return delta


class ServicioForm(forms.ModelForm):

        model = Servicio

        widgets = {"descripcion": forms.Textarea(attrs={"rows": 4})}

from django.core.files.storage import default_storage
from django.core.validators import MinValueValidator
from django.db import models
from django.db.models import Q

from apps.core.constraints import restriccion
from apps.core.fields import dinero
from apps.core.models.base import ModeloValidado
from apps.core.validators import validar_ruta_relativa, validar_sku


class Producto(ModeloValidado):
    sku = models.CharField(max_length=50, unique=True, validators=[validar_sku])
    nombre = models.CharField(max_length=120)
    descripcion = models.TextField(null=True, blank=True)
    precio = dinero()
    stock = models.IntegerField(default=0, validators=[MinValueValidator(0)])
    stock_minimo = models.IntegerField(default=0, validators=[MinValueValidator(0)])
    imagen = models.TextField(null=True, blank=True, validators=[validar_ruta_relativa])
    activo = models.BooleanField(default=True)
    creado_en = models.DateTimeField(auto_now_add=True)
    actualizado_en = models.DateTimeField(auto_now=True)

    campos_texto_opcional = ("descripcion", "imagen")

    class Meta:
        db_table = "productos"
        verbose_name = "producto"
        verbose_name_plural = "productos"
        constraints = [
            restriccion("ck_productos_precio", Q(precio__gte=0)),
            restriccion("ck_productos_stock", Q(stock__gte=0)),
            restriccion("ck_productos_stock_minimo", Q(stock_minimo__gte=0)),
        ]
        indexes = [models.Index(fields=["activo", "nombre"], name="idx_productos_activo_nombre")]

    def __str__(self):
        return f"{self.sku} - {self.nombre}"

    def clean(self):
        super().clean()
        if self.sku:
            self.sku = self.sku.strip().upper()

    @property
    def imagen_url(self):
        return default_storage.url(self.imagen) if self.imagen else ""

    @property
    def bajo_stock(self):
        return self.stock <= self.stock_minimo

    @property
    def disponibilidad_codigo(self):
        if self.stock <= 0:
            return "agotado"
        if self.stock <= self.stock_minimo:
            return "ultimas"
        return "disponible"

    @property
    def disponibilidad(self):
        return {
            "agotado": "Agotado",
            "ultimas": "Últimas unidades",
            "disponible": "Disponible",
        }[self.disponibilidad_codigo]

from decimal import Decimal

from django.core.exceptions import ValidationError
from django.core.validators import MinValueValidator
from django.db import models
from django.db.models import Q
from django.utils import timezone

from apps.core.constraints import en_lista, restriccion
from apps.core.models.base import ModeloValidado


class TipoDescuento(models.TextChoices):
    PORCENTAJE = "PORCENTAJE", "Porcentaje"
    MONTO_FIJO = "MONTO_FIJO", "Monto fijo"


class Promocion(ModeloValidado):
    nombre = models.CharField(max_length=120)
    descripcion = models.TextField(null=True, blank=True)
    tipo_descuento = models.CharField(max_length=20, choices=TipoDescuento.choices)
    valor_descuento = models.DecimalField(
        max_digits=12, decimal_places=2, validators=[MinValueValidator(Decimal("0.01"))]
    )
    fecha_inicio = models.DateTimeField()
    fecha_fin = models.DateTimeField()
    activo = models.BooleanField(default=True)

    campos_texto_opcional = ("descripcion",)

    class Meta:
        db_table = "promociones"
        verbose_name = "promoción"
        verbose_name_plural = "promociones"
        constraints = [
            en_lista("ck_promociones_tipo", "tipo_descuento", TipoDescuento.values),
            restriccion("ck_promociones_valor", Q(valor_descuento__gt=0)),
            restriccion("ck_promociones_fechas", Q(fecha_fin__gte=models.F("fecha_inicio"))),
            restriccion(
                "ck_promociones_porcentaje",
                ~Q(tipo_descuento="PORCENTAJE") | Q(valor_descuento__lte=100),
            ),
        ]
        indexes = [models.Index(fields=["activo", "fecha_inicio", "fecha_fin"], name="idx_promociones_vigencia")]

    def __str__(self):
        return self.nombre

    @property
    def vigente(self):
        return self.activo and self.fecha_inicio <= timezone.now() <= self.fecha_fin

    def clean(self):
        super().clean()
        errores = {}
        if self.fecha_inicio and self.fecha_fin and self.fecha_fin < self.fecha_inicio:
            errores["fecha_fin"] = "La fecha final no puede ser anterior a la inicial."
        if (
            self.tipo_descuento == TipoDescuento.PORCENTAJE
            and self.valor_descuento is not None
            and self.valor_descuento > 100
        ):
            errores["valor_descuento"] = "El porcentaje no puede superar 100."
        if errores:
            raise ValidationError(errores)

from django.db.models import F, Q

from apps.catalogo.models import CategoriaServicio, Producto, Servicio


def productos_publicos(busqueda=""):
    consulta = Producto.objects.filter(activo=True).order_by("nombre")
    if busqueda:
        consulta = consulta.filter(
            Q(nombre__icontains=busqueda) | Q(sku__icontains=busqueda) | Q(descripcion__icontains=busqueda)
        )
    return consulta


def productos_destacados(limite=4):
    return productos_publicos().filter(stock__gt=0)[:limite]


def servicios_publicos(categoria_id=None):
    consulta = Servicio.objects.filter(activo=True, categoria__activo=True).select_related("categoria")
    if categoria_id:
        consulta = consulta.filter(categoria_id=categoria_id)
    return consulta.order_by("categoria__nombre", "nombre")


def categorias_publicas():
    return CategoriaServicio.objects.filter(activo=True).order_by("nombre")


def productos_admin(busqueda="", solo_bajo_stock=False):
    consulta = Producto.objects.order_by("nombre")
    if busqueda:
        consulta = consulta.filter(Q(nombre__icontains=busqueda) | Q(sku__icontains=busqueda))
    if solo_bajo_stock:
        consulta = consulta.filter(stock__lte=F("stock_minimo"))
    return consulta

import uuid

from django.core.files.storage import default_storage
from django.db import transaction

from apps.catalogo.models import Producto


class ErrorInventario(Exception):
    pass


class StockInsuficiente(ErrorInventario):
    pass


class ProductoNoDisponible(ErrorInventario):
    pass


def _normalizar(items):
    normalizado = {}
    for producto_id, cantidad in dict(items).items():
        cantidad = int(cantidad)
        if cantidad <= 0:
            raise ValueError("La cantidad debe ser un entero positivo.")
        normalizado[int(producto_id)] = cantidad
    if not normalizado:
        raise ValueError("No se indicaron productos.")
    return normalizado


def _bloquear(ids):
    return {p.pk: p for p in Producto.objects.select_for_update().filter(pk__in=ids).order_by("pk")}


def _guardar(producto):
    producto.save(update_fields=["stock", "actualizado_en"])


@transaction.atomic
def descontar_stock(items):
    items = _normalizar(items)
    productos = _bloquear(items)
    for producto_id in sorted(items):
        producto = productos.get(producto_id)
        if producto is None or not producto.activo:
            raise ProductoNoDisponible(f"El producto {producto_id} no está disponible.")
        if producto.stock < items[producto_id]:
            raise StockInsuficiente(f"Existencias insuficientes de {producto.nombre}.")
    for producto_id, cantidad in items.items():
        productos[producto_id].stock -= cantidad
        _guardar(productos[producto_id])
    return list(productos.values())


@transaction.atomic
def reponer_stock(items):
    items = _normalizar(items)
    productos = _bloquear(items)
    faltantes = set(items) - set(productos)
    if faltantes:
        raise ProductoNoDisponible(f"Productos inexistentes: {sorted(faltantes)}.")
    for producto_id, cantidad in items.items():
        productos[producto_id].stock += cantidad
        _guardar(productos[producto_id])
    return list(productos.values())


@transaction.atomic
def ajustar_stock(producto_id, delta):
    delta = int(delta)
    producto = Producto.objects.select_for_update().get(pk=producto_id)
    nuevo = producto.stock + delta
    if nuevo < 0:
        raise StockInsuficiente("El ajuste dejaría el stock en negativo.")
    producto.stock = nuevo
    _guardar(producto)
    return producto


def guardar_imagen(archivo):
    extension = archivo.name.rsplit(".", 1)[-1].lower()
    return default_storage.save(f"productos/{uuid.uuid4().hex}.{extension}", archivo)


def eliminar_imagen(ruta):
    if ruta and default_storage.exists(ruta):
        default_storage.delete(ruta)

class TipoCliente(models.TextChoices):
    PERSONA = "PERSONA", "Persona"
    EMPRESA = "EMPRESA", "Empresa"


class Cliente(ModeloValidado):
    usuario = models.OneToOneField(
        settings.AUTH_USER_MODEL,
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        db_column="id_usuario",
        related_name="cliente",
    )
    tipo_cliente = models.CharField(max_length=20, choices=TipoCliente.choices)
    nombres = models.CharField(max_length=100, null=True, blank=True, validators=[validar_nombre_persona])
    apellidos = models.CharField(max_length=100, null=True, blank=True, validators=[validar_nombre_persona])
    nombre_empresa = models.CharField(max_length=150, null=True, blank=True)
    correo = models.EmailField(max_length=150)
    telefono = models.CharField(max_length=25, null=True, blank=True, validators=[validar_telefono])
    identificacion_fiscal = models.CharField(
        max_length=30, null=True, blank=True, validators=[validar_identificacion_fiscal]
    )
    direccion = models.CharField(max_length=250, null=True, blank=True)
    observaciones = models.TextField(null=True, blank=True)
    creado_en = models.DateTimeField(auto_now_add=True)
    actualizado_en = models.DateTimeField(auto_now=True)

    campos_texto_opcional = (
        "nombres",
        "apellidos",
        "nombre_empresa",
        "telefono",
        "identificacion_fiscal",
        "direccion",
        "observaciones",
    )

    class Meta:
        db_table = "clientes"
        verbose_name = "cliente"
        verbose_name_plural = "clientes"
        constraints = [en_lista("ck_clientes_tipo", "tipo_cliente", TipoCliente.values)]
        indexes = [
            models.Index(fields=["correo"], name="idx_clientes_correo"),
            models.Index(fields=["tipo_cliente"], name="idx_clientes_tipo"),
        ]

    def __str__(self):
        return self.nombre_visible

    @property
    def nombre_visible(self):
        if self.tipo_cliente == TipoCliente.EMPRESA:
            return self.nombre_empresa or self.correo
        return f"{self.nombres or ''} {self.apellidos or ''}".strip() or self.correo

    def clean(self):
        super().clean()
        errores = {}
        if self.correo:
            self.correo = self.correo.strip().lower()
        if self.tipo_cliente == TipoCliente.PERSONA:
            if not self.nombres:
                errores["nombres"] = "Los nombres son obligatorios para una persona."
            if not self.apellidos:
                errores["apellidos"] = "Los apellidos son obligatorios para una persona."
            if self.nombre_empresa:
                errores["nombre_empresa"] = "Una persona no puede tener nombre de empresa."
        elif self.tipo_cliente == TipoCliente.EMPRESA and not self.nombre_empresa:
            errores["nombre_empresa"] = "El nombre de la empresa es obligatorio."
        if errores:
            raise ValidationError(errores)

from apps.clientes.models import Cliente, TipoCliente


@transaction.atomic
def asegurar_cliente(usuario):
    cliente = Cliente.objects.filter(usuario=usuario).first()
    if cliente is not None:
        return cliente
    return Cliente.objects.create(
        usuario=usuario,
        tipo_cliente=TipoCliente.PERSONA,
        nombres=usuario.nombres,
        apellidos=usuario.apellidos,
        correo=usuario.correo,
        telefono=usuario.telefono,
    )


@transaction.atomic
def guardar_perfil(form):
    cliente = form.save()
    usuario = cliente.usuario
    if usuario is not None and cliente.tipo_cliente == TipoCliente.PERSONA:
        usuario.nombres = cliente.nombres
        usuario.apellidos = cliente.apellidos
        usuario.telefono = cliente.telefono
        usuario.save(update_fields=["nombres", "apellidos", "telefono"])
    return cliente

from decimal import ROUND_HALF_UP, Decimal, InvalidOperation

from django.conf import settings

from apps.core.constants import CENTAVO, TASA_IMPUESTO_DEFECTO


def tasa_impuesto():
    return Decimal(str(getattr(settings, "TASA_IMPUESTO", TASA_IMPUESTO_DEFECTO)))


def redondear(valor):
    return Decimal(valor).quantize(CENTAVO, rounding=ROUND_HALF_UP)


def subtotal_linea(cantidad, precio_unitario, descuento=Decimal("0")):
    return redondear(Decimal(cantidad) * Decimal(precio_unitario) - Decimal(descuento))


def calcular_linea(cantidad, precio_unitario, descuento=Decimal("0"), tasa=None):
    tasa = tasa_impuesto() if tasa is None else Decimal(str(tasa))
    descuento = redondear(descuento)
    subtotal = subtotal_linea(cantidad, precio_unitario, descuento)
    return {"subtotal": subtotal, "descuento": descuento, "impuesto": redondear(subtotal * tasa)}


def totales_documento(lineas):
    lineas = list(lineas)
    bruto = sum((linea["subtotal"] + linea["descuento"] for linea in lineas), Decimal("0"))
    descuento = sum((linea["descuento"] for linea in lineas), Decimal("0"))
    impuesto = sum((linea["impuesto"] for linea in lineas), Decimal("0"))
    return {
        "subtotal": redondear(bruto),
        "descuento": redondear(descuento),
        "impuesto": redondear(impuesto),
        "total": redondear(bruto - descuento + impuesto),
    }


def importes_coherentes(subtotal, descuento, impuesto, total):
    try:
        esperado = redondear(subtotal) - redondear(descuento) + redondear(impuesto)
        return esperado == redondear(total)
    except (InvalidOperation, TypeError, ValueError):
        return True


def linea_coherente(cantidad, precio_unitario, descuento, subtotal):
    try:
        return subtotal_linea(cantidad, precio_unitario, descuento) == redondear(subtotal)
    except (InvalidOperation, TypeError, ValueError):
        return True

from django.db import models


class ModeloValidado(models.Model):
    campos_texto_opcional = ()

    class Meta:
        abstract = True

    def normalizar(self):
        for campo in self.campos_texto_opcional:
            valor = getattr(self, campo)
            if isinstance(valor, str):
                setattr(self, campo, valor.strip() or None)

    def clean(self):
        super().clean()
        self.normalizar()

    def save(self, *args, **kwargs):
        update_fields = kwargs.get("update_fields")
        if update_fields is None:
            self.full_clean()
        else:
            nombres = set(update_fields)
            excluir = [
                campo.name
                for campo in self._meta.concrete_fields
                if campo.name not in nombres and campo.attname not in nombres
            ]
            self.clean_fields(exclude=excluir)
        super().save(*args, **kwargs)

def sembrar_datos_iniciales(sender=None, using="default", **kwargs):
    from apps.catalogo.models import CategoriaServicio
    from apps.pagos.models import MetodoPago
    from apps.usuarios.constants import GRUPOS
    from apps.usuarios.models import EstadoUsuario

    conexion = connections[using]
    existentes = set(conexion.introspection.table_names())
    requeridas = {modelo._meta.db_table for modelo in (EstadoUsuario, CategoriaServicio, MetodoPago, Group)}
    if not requeridas <= existentes:
        return

    with transaction.atomic(using=using):
        for id_, nombre, descripcion in ESTADOS_USUARIO:
            EstadoUsuario.objects.using(using).get_or_create(
                nombre=nombre, defaults={"id": id_, "descripcion": descripcion}
            )
        for id_, nombre, descripcion in METODOS_PAGO:
            MetodoPago.objects.using(using).get_or_create(
                nombre=nombre, defaults={"id": id_, "descripcion": descripcion}
            )
        for nombre, descripcion in CATEGORIAS_SERVICIOS:
            CategoriaServicio.objects.using(using).get_or_create(
                nombre=nombre, defaults={"descripcion": descripcion}
            )
        for nombre in GRUPOS:
            Group.objects.using(using).get_or_create(name=nombre)

        sentencias = conexion.ops.sequence_reset_sql(no_style(), [EstadoUsuario, MetodoPago])
        with conexion.cursor() as cursor:
            for sentencia in sentencias:
                cursor.execute(sentencia)

@register.simple_tag
def whatsapp_url(telefono, mensaje=""):
    digitos = re.sub(r"\D", "", str(telefono or ""))
    if not digitos:
        return ""
    enlace = f"https://wa.me/{digitos}"
    return f"{enlace}?text={quote(str(mensaje))}" if mensaje else enlace

def validar_ruta_relativa(valor):
    texto = str(valor)
    partes = re.split(r"[\\/]", texto)
    if (
        not texto
        or len(texto) > 500
        or texto != texto.strip()
        or texto.startswith(("/", "\\", "~"))
        or re.match(r"^[A-Za-z]:", texto)
        or "://" in texto
        or any(parte in ("", ".", "..") for parte in partes)
        or any(ord(caracter) < 32 for caracter in texto)
    ):
        raise ValidationError("Ruta de archivo no permitida.", code="ruta_invalida")


def validar_nombre_archivo(valor):
    texto = str(valor)
    if (
        not texto
        or texto != texto.strip()
        or texto.startswith(".")
        or re.search(r"[\\/\x00-\x1f]", texto)
    ):
        raise ValidationError("Nombre de archivo no permitido.", code="archivo_invalido")


def validar_tamano_archivo(valor):
    maximo = getattr(settings, "MAX_ARCHIVO_BYTES", MAX_ARCHIVO_BYTES)
    if valor > maximo:
        raise ValidationError(f"El archivo supera el maximo permitido de {maximo} bytes.", code="archivo_grande")


def validar_objeto_json(valor):
    if valor is None:
        return
    if not isinstance(valor, dict):
        raise ValidationError("Debe ser un objeto JSON.", code="json_invalido")
    try:
        longitud = len(json.dumps(valor, ensure_ascii=False))
    except (TypeError, ValueError):
        raise ValidationError("El objeto JSON no es serializable.", code="json_invalido")
    if longitud > MAX_JSON_BYTES:
        raise ValidationError("El objeto JSON es demasiado grande.", code="json_grande")

        form = AjusteStockForm(request.POST)
        if not form.is_valid():
            messages.error(request, "Ingrese un ajuste de stock válido.")
            return redirect("administrator:inventario")

            producto = servicio_catalogo.ajustar_stock(pk, form.cleaned_data["delta"])

            raise Http404
        except servicio_catalogo.ErrorInventario as error:
            messages.error(request, str(error))
        else:
            messages.success(request, f"Stock de {producto.nombre}: {producto.stock}.")
        return redirect("administrator:inventario")

class CuentaView(ClienteRequeridoMixin, View):

            return redirect("customer:cuenta")
        return render(request, self.template_name, self._contexto(form, cliente))

from django.utils.http import url_has_allowed_host_and_scheme
from django.views.generic import TemplateView


class AccesoView(TemplateView):
    template_name = "guest/acceso.html"

    def dispatch(self, request, *args, **kwargs):
        if request.user.is_authenticated:
            return redirect("redirigir")
        return super().dispatch(request, *args, **kwargs)

    def get_context_data(self, **kwargs):
        contexto = super().get_context_data(**kwargs)
        siguiente = self.request.GET.get("next", "")
        if not url_has_allowed_host_and_scheme(siguiente, {self.request.get_host()}, self.request.is_secure()):
            siguiente = ""
        contexto["siguiente"] = siguiente
        return contexto

def _nombre_o_defecto(valor, defecto):
    valor = (valor or "").strip()
    try:
        validar_nombre_persona(valor)
    except ValidationError:
        return defecto
    return valor


class CuentaAdapter(DefaultAccountAdapter):
    def is_open_for_signup(self, request):
        return True

    def save_user(self, request, user, form, commit=True):
        datos = form.cleaned_data
        correo = datos.get("email")
        if correo:
            user.correo = correo.strip().lower()
        if datos.get("nombres"):
            user.nombres = datos["nombres"].strip()
        if datos.get("apellidos"):
            user.apellidos = datos["apellidos"].strip()
        if "telefono" in datos:
            user.telefono = datos["telefono"] or None
        user.is_staff = False
        user.is_superuser = False
        if getattr(form, "sociallogin", None) is None:
            user.estado_id = ESTADO_PENDIENTE
        password = datos.get("password1")
        if password:
            user.set_password(password)
        else:
            user.set_unusable_password()
        if commit:
            user.save()
        return user


class SocialAdapter(DefaultSocialAccountAdapter):
    def is_open_for_signup(self, request, sociallogin):
        return True

from django.contrib.auth.models import BaseUserManager

from apps.usuarios.constants import ESTADO_ACTIVO


class UsuarioManager(BaseUserManager):
    def get_by_natural_key(self, correo):
        return self.get(**{f"{self.model.USERNAME_FIELD}__iexact": correo})

    def _crear(self, correo, password, **extra):
        if not correo:
            raise ValueError("El correo es obligatorio.")
        extra.setdefault("estado_id", ESTADO_ACTIVO)
        usuario = self.model(correo=self.normalize_email(correo).lower(), **extra)
        if password:
            usuario.set_password(password)
        else:
            usuario.set_unusable_password()
        usuario.save(using=self._db)
        return usuario

    def create_user(self, correo, password=None, **extra):
        extra.setdefault("is_staff", False)
        extra.setdefault("is_superuser", False)
        return self._crear(correo, password, **extra)

    def create_superuser(self, correo, password=None, **extra):
        extra.setdefault("is_staff", True)
        extra.setdefault("is_superuser", True)
        if extra.get("is_staff") is not True or extra.get("is_superuser") is not True:
            raise ValueError("El superusuario debe tener is_staff e is_superuser en True.")
        return self._crear(correo, password, **extra)

from django.contrib.auth.models import AbstractBaseUser, PermissionsMixin
from django.db import models
from django.utils import timezone

from apps.core.models.base import ModeloValidado
from apps.core.validators import validar_nombre_persona, validar_telefono
from apps.usuarios.constants import ESTADO_ACTIVO, ESTADO_INACTIVO, ESTADOS_BLOQUEADOS
from apps.usuarios.managers import UsuarioManager


class Usuario(ModeloValidado, AbstractBaseUser, PermissionsMixin):
    nombres = models.CharField(max_length=100, validators=[validar_nombre_persona])
    apellidos = models.CharField(max_length=100, validators=[validar_nombre_persona])
    correo = models.EmailField(max_length=150, unique=True)
    telefono = models.CharField(max_length=25, null=True, blank=True, validators=[validar_telefono])
    is_staff = models.BooleanField(default=False)
    is_active = models.BooleanField(default=True)
    date_joined = models.DateTimeField(default=timezone.now)
    estado = models.ForeignKey(
        "usuarios.EstadoUsuario",
        on_delete=models.PROTECT,
        db_column="id_estado",
        related_name="usuarios",
        default=ESTADO_ACTIVO,
    )

    objects = UsuarioManager()

    USERNAME_FIELD = "correo"
    EMAIL_FIELD = "correo"
    REQUIRED_FIELDS = ["nombres", "apellidos"]
    campos_texto_opcional = ("telefono",)

    class Meta:
        db_table = "usuarios"
        verbose_name = "usuario"
        verbose_name_plural = "usuarios"

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self._estado_inicial = self.__dict__.get("estado_id")

    def __str__(self):
        return self.correo

    @property
    def nombre_completo(self):
        return f"{self.nombres} {self.apellidos}".strip()

    def get_full_name(self):
        return self.nombre_completo

    def get_short_name(self):
        return self.nombres

    def clean(self):
        super().clean()
        if self.correo:
            self.correo = self.correo.strip().lower()
        if self.estado_id in ESTADOS_BLOQUEADOS:
            self.is_active = False
        elif self.estado_id == ESTADO_ACTIVO:
            if self._estado_inicial in ESTADOS_BLOQUEADOS:
                self.is_active = True
            elif not self.is_active:
                self.estado_id = ESTADO_INACTIVO

    def save(self, *args, **kwargs):
        super().save(*args, **kwargs)
        self._estado_inicial = self.estado_id

    def tiene_algun_rol(self, roles):
        if not (self.is_active and self.estado_id == ESTADO_ACTIVO):
            return False
        if self.is_superuser:
            return True
        return self.groups.filter(name__in=roles).exists()

def obtener_identificador(request, credentials=None):
    credentials = credentials or {}
    for clave in ("correo", "email", "login", "username"):
        valor = credentials.get(clave)
        if not valor and request is not None and hasattr(request, "POST"):
            valor = request.POST.get(clave)
        if valor:
            return str(valor).strip().lower()
    return None

#!/usr/bin/env bash

set -Eeuo pipefail

# ============================================================
# INSPECTOR UNIVERSAL DE PROYECTOS
# Compatible con proyectos de cualquier lenguaje o framework.
# ============================================================

# ---------------------- CONFIGURACIÓN -----------------------

readonly NOMBRE_SALIDA="project-context.txt"

# Tamaño máximo por archivo: 1 MB por defecto.
# Se puede modificar mediante la variable MAX_FILE_SIZE.
MAX_FILE_SIZE="${MAX_FILE_SIZE:-1048576}"

# Directorios que normalmente contienen dependencias,
# archivos temporales, cachés o resultados de compilación.
readonly -a EXCLUDE_DIRS=(
    ".git"
    ".svn"
    ".hg"
    "node_modules"
    "vendor"
    ".venv"
    "venv"
    "__pycache__"
    ".pytest_cache"
    ".mypy_cache"
    ".ruff_cache"
    ".tox"
    ".cache"
    ".idea"
    ".vscode"
    ".angular"
    ".next"
    ".nuxt"
    ".svelte-kit"
    ".turbo"
    ".gradle"
    ".terraform"
    ".dart_tool"
    "Pods"
    "DerivedData"
    "coverage"
    "dist"
    "build"
    "target"
    "out"
    "bin"
    "obj"
)

# Patrones de archivos que pueden contener información sensible.
readonly -a SENSITIVE_PATTERNS=(
    ".env"
    ".env.*"
    "*.pem"
    "*.key"
    "*.p12"
    "*.pfx"
    "id_rsa*"
    "id_ed25519*"
    ".netrc"
    ".npmrc"
    ".pypirc"
    "credentials.json"
    "*credentials*.json"
    "*secrets*.json"
    "*.tfstate"
    "*.tfstate.backup"
)

# ---------------------- ARGUMENTOS --------------------------

mostrar_ayuda() {
    cat <<'EOF'
INSPECTOR UNIVERSAL DE PROYECTOS

Uso:
  ./inspeccionar-proyecto.sh [DIRECTORIO] [ARCHIVO_SALIDA]

Ejemplos:
  ./inspeccionar-proyecto.sh
  ./inspeccionar-proyecto.sh ./backend
  ./inspeccionar-proyecto.sh ./backend informe.txt
  ./inspeccionar-proyecto.sh ~/Projects/mi-proyecto ~/Documents/informe.txt

Variables de entorno:
  MAX_FILE_SIZE=2097152

La variable MAX_FILE_SIZE permite modificar el tamaño máximo
de cada archivo que se incorporará al informe (en bytes).

EOF
}

if [[ "${1:-}" == "-h" || "${1:-}" == "--help" ]]; then
    mostrar_ayuda
    exit 0
fi

if (( $# > 2 )); then
    echo "[ERROR] Número de argumentos incorrecto."
    mostrar_ayuda
    exit 1
fi

# ---------------------- VALIDACIONES ------------------------

if [[ ! "$MAX_FILE_SIZE" =~ ^[0-9]+$ ]] || (( MAX_FILE_SIZE < 1 )); then
    echo "[ERROR] MAX_FILE_SIZE debe ser un entero positivo."
    exit 1
fi

DIRECTORIO="${1:-.}"
SALIDA="${2:-$NOMBRE_SALIDA}"

if [[ ! -d "$DIRECTORIO" ]]; then
    echo "[ERROR] El directorio indicado no existe: $DIRECTORIO"
    exit 1
fi

ROOT="$(cd -- "$DIRECTORIO" && pwd -P)"

# Resolver la ruta de salida.
if [[ "$SALIDA" != /* ]]; then
    SALIDA="$ROOT/$SALIDA"
fi

DIRECTORIO_SALIDA="$(dirname -- "$SALIDA")"
NOMBRE_ARCHIVO="$(basename -- "$SALIDA")"

mkdir -p -- "$DIRECTORIO_SALIDA"

DIRECTORIO_SALIDA="$(cd -- "$DIRECTORIO_SALIDA" && pwd -P)"
SALIDA="$DIRECTORIO_SALIDA/$NOMBRE_ARCHIVO"

if [[ "$SALIDA" == "$ROOT" ]]; then
    echo "[ERROR] La ruta de salida no puede ser un directorio."
    exit 1
fi

if [[ -d "$SALIDA" ]]; then
    echo "[ERROR] La ruta de salida corresponde a un directorio."
    exit 1
fi

# ---------------------- FUNCIONES ---------------------------

es_archivo_sensible() {
    local nombre
    nombre="$(basename -- "$1")"

    for patron in "${SENSITIVE_PATTERNS[@]}"; do
        if [[ "$nombre" == $patron ]]; then
            return 0
        fi
    done

    return 1
}

buscar_elementos() {
    local -a argumentos

    argumentos=("$ROOT" "(" -type d "(")

    for directorio in "${EXCLUDE_DIRS[@]}"; do
        argumentos+=(-name "$directorio" -o)
    done

    # -false evita dejar un -o al final de la expresión.
    argumentos+=(
        -false
        ")"
        -prune
        ")"
        -o
        "("
        -type f
        -o
        -type d
        ")"
        -print0
    )

    find "${argumentos[@]}"
}

obtener_ruta_relativa() {
    local ruta="$1"

    if [[ "$ROOT" == "/" ]]; then
        printf '%s' "${ruta#/}"
    else
        printf '%s' "${ruta#"$ROOT"/}"
    fi
}

es_archivo_texto() {
    local archivo="$1"

    # Los archivos vacíos también son válidos.
    [[ ! -s "$archivo" ]] && return 0

    # grep -Iq permite detectar contenido binario habitual.
    grep -Iq . "$archivo"
}

# ---------------------- INICIALIZACIÓN ----------------------

echo
echo "============================================================"
echo "             INSPECTOR UNIVERSAL DE PROYECTOS"
echo "============================================================"
echo
echo "Directorio raíz : $ROOT"
echo "Archivo salida  : $SALIDA"
echo "Límite por archivo: $MAX_FILE_SIZE bytes"
echo

if [[ -e "$SALIDA" && ! -w "$SALIDA" ]]; then
    echo "[ERROR] No tienes permisos para sobrescribir el informe."
    exit 1
fi

echo "[INFO] Explorando directorios y archivos..."

mapfile -d '' -t ELEMENTOS < <(
    buscar_elementos | LC_ALL=C sort -z
)

# ---------------------- GENERACIÓN --------------------------

TOTAL_ARCHIVOS=0
ARCHIVOS_INCLUIDOS=0
ARCHIVOS_OMITIDOS=0

{
    echo "============================================================"
    echo "              INFORME DEL PROYECTO"
    echo "============================================================"
    echo
    echo "Directorio raíz: $ROOT"
    echo "Fecha: $(date '+%Y-%m-%d %H:%M:%S')"
    echo "Límite por archivo: $MAX_FILE_SIZE bytes"
    echo

    # ========================================================
    # 1. ESTRUCTURA DEL PROYECTO
    # ========================================================

    echo "============================================================"
    echo " 1. ESTRUCTURA DE DIRECTORIOS Y ARCHIVOS"
    echo "============================================================"
    echo

    printf '[ROOT] %s/\n' "$(basename -- "$ROOT")"

    for elemento in "${ELEMENTOS[@]}"; do

        [[ "$elemento" == "$ROOT" ]] && continue
        [[ "$elemento" == "$SALIDA" ]] && continue

        relativo="$(obtener_ruta_relativa "$elemento")"

        separadores="${relativo//[^\/]/}"
        nivel="${#separadores}"

        if [[ -d "$elemento" ]]; then
            printf '%*s[DIR]  %s/\n' \
                "$((nivel * 4))" "" \
                "$(basename -- "$elemento")"

        elif [[ -f "$elemento" ]]; then
            printf '%*s[FILE] %s\n' \
                "$((nivel * 4))" "" \
                "$(basename -- "$elemento")"
        fi
    done

    echo

    # ========================================================
    # 2. CONTENIDO DE LOS ARCHIVOS
    # ========================================================

    echo "============================================================"
    echo " 2. CONTENIDO DE LOS ARCHIVOS"
    echo "============================================================"
    echo

    for elemento in "${ELEMENTOS[@]}"; do

        [[ ! -f "$elemento" ]] && continue
        [[ "$elemento" == "$SALIDA" ]] && continue

        ((TOTAL_ARCHIVOS += 1))

        relativo="$(obtener_ruta_relativa "$elemento")"

        echo "============================================================"
        echo "ARCHIVO: $relativo"
        echo "============================================================"
        echo

        # Comprobar información sensible.
        if es_archivo_sensible "$elemento"; then
            echo "[ARCHIVO OMITIDO: posible información sensible]"
            ((ARCHIVOS_OMITIDOS += 1))
            echo
            continue
        fi

        # Comprobar tamaño.
        tamano="$(wc -c < "$elemento")"

        if (( tamano > MAX_FILE_SIZE )); then
            echo "[ARCHIVO OMITIDO: supera el límite permitido]"
            echo "Tamaño: $tamano bytes"
            ((ARCHIVOS_OMITIDOS += 1))
            echo
            continue
        fi

        # Comprobar contenido binario.
        if ! es_archivo_texto "$elemento"; then
            echo "[ARCHIVO OMITIDO: contenido binario]"
            ((ARCHIVOS_OMITIDOS += 1))
            echo
            continue
        fi

        # Incorporar contenido.
        if [[ -s "$elemento" ]]; then
            cat -- "$elemento"
        else
            echo "[Archivo vacío]"
        fi

        ((ARCHIVOS_INCLUIDOS += 1))

        echo
        echo
    done

    # ========================================================
    # 3. RESUMEN
    # ========================================================

    echo "============================================================"
    echo " 3. RESUMEN DE LA INSPECCIÓN"
    echo "============================================================"
    echo
    echo "Archivos encontrados: $TOTAL_ARCHIVOS"
    echo "Archivos incluidos:   $ARCHIVOS_INCLUIDOS"
    echo "Archivos omitidos:    $ARCHIVOS_OMITIDOS"
    echo

} > "$SALIDA"

# ---------------------- FINALIZACIÓN ------------------------

echo "[OK] Inspección completada."
echo
echo "Archivos encontrados: $TOTAL_ARCHIVOS"
echo "Archivos incluidos:   $ARCHIVOS_INCLUIDOS"
echo "Archivos omitidos:    $ARCHIVOS_OMITIDOS"
echo
echo "Informe generado:"
echo "$SALIDA"
echo
du -h -- "$SALIDA"
echo
echo "============================================================"
