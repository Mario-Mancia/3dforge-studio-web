from django import forms
from django.conf import settings
from django.core.exceptions import ValidationError

from apps.catalogo.models import CategoriaServicio, Producto, Servicio
from apps.catalogo.services.catalogo import eliminar_imagen, guardar_imagen
from apps.core.constants import FORMATOS_IMAGEN, MAX_IMAGEN_BYTES

FORMATOS_PILLOW = {"JPEG", "PNG", "WEBP"}


class ProductoForm(forms.ModelForm):
    imagen_archivo = forms.ImageField(required=False, label="Imagen")
    quitar_imagen = forms.BooleanField(required=False, label="Quitar imagen actual")

    class Meta:
        model = Producto
        fields = ["sku", "nombre", "descripcion", "precio", "stock", "stock_minimo", "activo"]
        widgets = {"descripcion": forms.Textarea(attrs={"rows": 4})}

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        if self.instance.pk:
            self.fields.pop("stock")

    def clean_imagen_archivo(self):
        archivo = self.cleaned_data.get("imagen_archivo")
        if not archivo:
            return archivo
        extension = archivo.name.rsplit(".", 1)[-1].lower() if "." in archivo.name else ""
        if extension not in FORMATOS_IMAGEN:
            raise ValidationError("Formato no permitido. Use JPG, PNG o WEBP.")
        maximo = getattr(settings, "IMAGEN_MAX_BYTES", MAX_IMAGEN_BYTES)
        if archivo.size > maximo:
            raise ValidationError("La imagen supera el tamaño máximo permitido.")
        imagen = getattr(archivo, "image", None)
        if imagen is None or imagen.format not in FORMATOS_PILLOW:
            raise ValidationError("El archivo no es una imagen válida.")
        return archivo

    def save(self, commit=True):
        producto = super().save(commit=False)
        archivo = self.cleaned_data.get("imagen_archivo")
        anterior = producto.imagen
        nueva = None
        if archivo:
            nueva = guardar_imagen(archivo)
            producto.imagen = nueva
        elif self.cleaned_data.get("quitar_imagen"):
            producto.imagen = None
        if commit:
            try:
                producto.save()
            except Exception:
                if nueva:
                    eliminar_imagen(nueva)
                raise
            if anterior and anterior != producto.imagen:
                eliminar_imagen(anterior)
        return producto


class AjusteStockForm(forms.Form):
    delta = forms.IntegerField(min_value=-100000, max_value=100000, label="Ajuste")

    def clean_delta(self):
        delta = self.cleaned_data["delta"]
        if delta == 0:
            raise ValidationError("El ajuste no puede ser cero.")
        return delta

from decimal import Decimal

from django.conf import settings
from django.core.management.base import BaseCommand, CommandError
from django.db import transaction

from apps.catalogo.models import CategoriaServicio, Producto, Servicio

PRODUCTOS = (
    ("DEMO-001", "Soporte articulado para celular", "Soporte impreso en PLA, ajustable.", "6.50", 25, 5),
    ("DEMO-002", "Organizador de escritorio", "Organizador modular para lápices y notas.", "9.90", 12, 4),
    ("DEMO-003", "Maceta geométrica", "Maceta decorativa con diseño low-poly.", "7.25", 4, 5),
    ("DEMO-004", "Llavero personalizable", "Llavero con iniciales en relieve.", "2.50", 60, 10),
    ("DEMO-005", "Figura articulada", "Figura coleccionable con partes móviles.", "14.00", 0, 3),
    ("DEMO-006", "Porta audífonos", "Soporte de escritorio para audífonos.", "8.75", 18, 5),
)

SERVICIOS = (
    ("Impresión 3D", "Impresión por pieza", "Impresión de su modelo en el material elegido.", "12.00", True),
    ("Diseño 3D", "Diseño de pieza a medida", "Diseño de una pieza a partir de sus medidas.", "35.00", False),
    ("Modelado 3D", "Modelado desde boceto", "Modelado tridimensional a partir de un boceto.", "28.00", True),
    ("Asesorías", "Asesoría técnica", "Asesoría para preparar su proyecto de impresión.", "20.00", False),
    ("Prototipado", "Prototipo funcional", "Prototipo para validar un diseño antes de producir.", "45.00", True),
)


class Command(BaseCommand):
    help = "Crea productos y servicios de ejemplo (idempotente)."

    def add_arguments(self, parser):
        parser.add_argument("--forzar", action="store_true", help="Permite ejecutarlo con DEBUG=False.")

    @transaction.atomic
    def handle(self, *args, **opciones):
        if not settings.DEBUG and not opciones["forzar"]:
            raise CommandError("Solo se ejecuta con DEBUG=True (o use --forzar).")
        for sku, nombre, descripcion, precio, stock, minimo in PRODUCTOS:
            Producto.objects.get_or_create(
                sku=sku,
                defaults={
                    "nombre": nombre,
                    "descripcion": descripcion,
                    "precio": Decimal(precio),
                    "stock": stock,
                    "stock_minimo": minimo,
                },
            )
        for categoria, nombre, descripcion, precio, requiere in SERVICIOS:
            categoria_obj, _ = CategoriaServicio.objects.get_or_create(nombre=categoria)
            Servicio.objects.get_or_create(
                categoria=categoria_obj,
                nombre=nombre,
                defaults={
                    "descripcion": descripcion,
                    "precio_base": Decimal(precio),
                    "requiere_archivo": requiere,
                },
            )
        self.stdout.write(self.style.SUCCESS("Datos de ejemplo listos."))

from apps.core.constraints import restriccion
from apps.core.fields import dinero
from apps.core.models.base import ModeloValidado
from apps.core.validators import validar_ruta_relativa, validar_sku


class Producto(ModeloValidado):
    sku = models.CharField(max_length=50, unique=True, validators=[validar_sku])
    nombre = models.CharField(max_length=120)
    descripcion = models.TextField(null=True, blank=True)
    precio = dinero()
    stock = models.IntegerField(default=0, validators=[MinValueValidator(0)])
    stock_minimo = models.IntegerField(default=0, validators=[MinValueValidator(0)])
    imagen = models.TextField(null=True, blank=True, validators=[validar_ruta_relativa])
    activo = models.BooleanField(default=True)
    creado_en = models.DateTimeField(auto_now_add=True)
    actualizado_en = models.DateTimeField(auto_now=True)

    campos_texto_opcional = ("descripcion", "imagen")

    class Meta:
        db_table = "productos"
        verbose_name = "producto"
        verbose_name_plural = "productos"
        constraints = [
            restriccion("ck_productos_precio", Q(precio__gte=0)),
            restriccion("ck_productos_stock", Q(stock__gte=0)),
            restriccion("ck_productos_stock_minimo", Q(stock_minimo__gte=0)),
        ]
        indexes = [models.Index(fields=["activo", "nombre"], name="idx_productos_activo_nombre")]

    def __str__(self):
        return f"{self.sku} - {self.nombre}"

    def clean(self):
        super().clean()
        if self.sku:
            self.sku = self.sku.strip().upper()

    @property
    def imagen_url(self):
        return default_storage.url(self.imagen) if self.imagen else ""

    @property
    def bajo_stock(self):
        return self.stock <= self.stock_minimo

    @property
    def disponibilidad_codigo(self):
        if self.stock <= 0:
            return "agotado"
        if self.stock <= self.stock_minimo:
            return "ultimas"
        return "disponible"

    @property
    def disponibilidad(self):
        return {
            "agotado": "Agotado",
            "ultimas": "Últimas unidades",
            "disponible": "Disponible",
        }[self.disponibilidad_codigo]

class TipoDescuento(models.TextChoices):
    PORCENTAJE = "PORCENTAJE", "Porcentaje"
    MONTO_FIJO = "MONTO_FIJO", "Monto fijo"


class Promocion(ModeloValidado):
    nombre = models.CharField(max_length=120)
    descripcion = models.TextField(null=True, blank=True)
    tipo_descuento = models.CharField(max_length=20, choices=TipoDescuento.choices)
    valor_descuento = models.DecimalField(
        max_digits=12, decimal_places=2, validators=[MinValueValidator(Decimal("0.01"))]
    )
    fecha_inicio = models.DateTimeField()
    fecha_fin = models.DateTimeField()
    activo = models.BooleanField(default=True)

    campos_texto_opcional = ("descripcion",)

    class Meta:
        db_table = "promociones"
        verbose_name = "promoción"
        verbose_name_plural = "promociones"
        constraints = [
            en_lista("ck_promociones_tipo", "tipo_descuento", TipoDescuento.values),
            restriccion("ck_promociones_valor", Q(valor_descuento__gt=0)),
            restriccion("ck_promociones_fechas", Q(fecha_fin__gte=models.F("fecha_inicio"))),
            restriccion(
                "ck_promociones_porcentaje",
                ~Q(tipo_descuento="PORCENTAJE") | Q(valor_descuento__lte=100),
            ),
        ]
        indexes = [models.Index(fields=["activo", "fecha_inicio", "fecha_fin"], name="idx_promociones_vigencia")]

    def __str__(self):
        return self.nombre

    @property
    def vigente(self):
        return self.activo and self.fecha_inicio <= timezone.now() <= self.fecha_fin

    def clean(self):
        super().clean()
        errores = {}
        if self.fecha_inicio and self.fecha_fin and self.fecha_fin < self.fecha_inicio:
            errores["fecha_fin"] = "La fecha final no puede ser anterior a la inicial."
        if (
            self.tipo_descuento == TipoDescuento.PORCENTAJE
            and self.valor_descuento is not None
            and self.valor_descuento > 100
        ):
            errores["valor_descuento"] = "El porcentaje no puede superar 100."
        if errores:
            raise ValidationError(errores)

    return consulta

    if categoria_id:
        consulta = consulta.filter(categoria_id=categoria_id)
    return consulta.order_by("categoria__nombre", "nombre")


def categorias_publicas():
    return CategoriaServicio.objects.filter(activo=True).order_by("nombre")

    consulta = Producto.objects.order_by("nombre")
    if busqueda:
        consulta = consulta.filter(Q(nombre__icontains=busqueda) | Q(sku__icontains=busqueda))
    if solo_bajo_stock:
        consulta = consulta.filter(stock__lte=F("stock_minimo"))

import uuid

from django.core.files.storage import default_storage
from django.db import transaction

from apps.catalogo.models import Producto


class ErrorInventario(Exception):
    pass


class StockInsuficiente(ErrorInventario):
    pass


class ProductoNoDisponible(ErrorInventario):
    pass


def _normalizar(items):
    normalizado = {}
    for producto_id, cantidad in dict(items).items():
        cantidad = int(cantidad)
        if cantidad <= 0:
            raise ValueError("La cantidad debe ser un entero positivo.")
        normalizado[int(producto_id)] = cantidad
    if not normalizado:
        raise ValueError("No se indicaron productos.")
    return normalizado


def _bloquear(ids):
    return {p.pk: p for p in Producto.objects.select_for_update().filter(pk__in=ids).order_by("pk")}


def _guardar(producto):
    producto.save(update_fields=["stock", "actualizado_en"])


@transaction.atomic
def descontar_stock(items):
    items = _normalizar(items)
    productos = _bloquear(items)
    for producto_id in sorted(items):
        producto = productos.get(producto_id)
        if producto is None or not producto.activo:
            raise ProductoNoDisponible(f"El producto {producto_id} no está disponible.")
        if producto.stock < items[producto_id]:
            raise StockInsuficiente(f"Existencias insuficientes de {producto.nombre}.")
    for producto_id, cantidad in items.items():
        productos[producto_id].stock -= cantidad
        _guardar(productos[producto_id])
    return list(productos.values())


@transaction.atomic
def reponer_stock(items):
    items = _normalizar(items)
    productos = _bloquear(items)
    faltantes = set(items) - set(productos)
    if faltantes:
        raise ProductoNoDisponible(f"Productos inexistentes: {sorted(faltantes)}.")
    for producto_id, cantidad in items.items():
        productos[producto_id].stock += cantidad
        _guardar(productos[producto_id])
    return list(productos.values())


@transaction.atomic
def ajustar_stock(producto_id, delta):
    delta = int(delta)
    producto = Producto.objects.select_for_update().get(pk=producto_id)
    nuevo = producto.stock + delta
    if nuevo < 0:
        raise StockInsuficiente("El ajuste dejaría el stock en negativo.")
    producto.stock = nuevo
    _guardar(producto)
    return producto


def guardar_imagen(archivo):
    extension = archivo.name.rsplit(".", 1)[-1].lower()
    return default_storage.save(f"productos/{uuid.uuid4().hex}.{extension}", archivo)


def eliminar_imagen(ruta):
    if ruta and default_storage.exists(ruta):
        default_storage.delete(ruta)

from django.db import transaction

from apps.clientes.models import Cliente, TipoCliente


@transaction.atomic
def asegurar_cliente(usuario):
    cliente = Cliente.objects.filter(usuario=usuario).first()
    if cliente is not None:
        return cliente
    return Cliente.objects.create(
        usuario=usuario,
        tipo_cliente=TipoCliente.PERSONA,
        nombres=usuario.nombres,
        apellidos=usuario.apellidos,
        correo=usuario.correo,
        telefono=usuario.telefono,
    )


@transaction.atomic
def guardar_perfil(form):
    cliente = form.save()
    usuario = cliente.usuario
    if usuario is not None and cliente.tipo_cliente == TipoCliente.PERSONA:
        usuario.nombres = cliente.nombres
        usuario.apellidos = cliente.apellidos
        usuario.telefono = cliente.telefono
        usuario.save(update_fields=["nombres", "apellidos", "telefono"])
    return cliente

import os
import re
import struct

from django.conf import settings
from django.core.exceptions import ValidationError
from django.http import FileResponse, Http404

from apps.core.constants import FORMATOS_ARCHIVO, MAX_ARCHIVO_BYTES

EXTENSIONES = {
    "stl": "STL",
    "obj": "OBJ",
    "3mf": "3MF",
    "step": "STEP",
    "stp": "STEP",
    "png": "PNG",
    "jpg": "JPG",
    "jpeg": "JPG",
    "pdf": "PDF",
}

TEXTO_PLANO = re.compile(rb"[\x09\x0a\x0d\x20-\x7e\x80-\xff]*")


def nombre_seguro(nombre):
    base = os.path.basename(str(nombre).replace("\\", "/"))
    base = re.sub(r"[^\w.\- ]", "_", base, flags=re.UNICODE).strip(" .")
    return base[:150]


def _leer_cabecera(archivo, tamano=1024):
    archivo.seek(0)
    datos = archivo.read(tamano)
    archivo.seek(0)
    return datos


def _stl_valido(archivo, cabecera):
    if cabecera.startswith(b"solid") and b"\x00" not in cabecera:
        return True
    if archivo.size < 84:
        return False
    archivo.seek(80)
    triangulos = struct.unpack("<I", archivo.read(4))[0]
    archivo.seek(0)
    return archivo.size == 84 + 50 * triangulos


def _contenido_coincide(formato, archivo, cabecera):
    if formato == "STL":
        return _stl_valido(archivo, cabecera)
    if formato == "OBJ":
        return bool(TEXTO_PLANO.fullmatch(cabecera))
    if formato == "3MF":
        return cabecera.startswith(b"PK\x03\x04")
    if formato == "STEP":
        return cabecera.lstrip().startswith(b"ISO-10303-21")
    if formato == "PNG":
        return cabecera.startswith(b"\x89PNG\r\n\x1a\n")
    if formato == "JPG":
        return cabecera.startswith(b"\xff\xd8\xff")
    if formato == "PDF":
        return cabecera.startswith(b"%PDF-")
    return False


def validar_archivo(archivo):
    nombre = nombre_seguro(archivo.name)
    if not nombre or "." not in nombre:
        raise ValidationError("El archivo debe tener una extensión.", code="archivo_sin_extension")
    formato = EXTENSIONES.get(nombre.rsplit(".", 1)[-1].lower())
    if formato is None:
        raise ValidationError(
            f"Formato no permitido. Formatos válidos: {', '.join(FORMATOS_ARCHIVO)}.", code="formato_no_permitido"
        )
    if archivo.size <= 0:
        raise ValidationError("El archivo está vacío.", code="archivo_vacio")
    maximo = getattr(settings, "MAX_ARCHIVO_BYTES", MAX_ARCHIVO_BYTES)
    if archivo.size > maximo:
        raise ValidationError("El archivo supera el tamaño máximo permitido.", code="archivo_grande")
    if not _contenido_coincide(formato, archivo, _leer_cabecera(archivo)):
        raise ValidationError("El contenido del archivo no corresponde a su formato.", code="contenido_invalido")
    return formato, nombre


def respuesta_descarga(almacenamiento, ruta, nombre):
    if not ruta or not almacenamiento.exists(ruta):
        raise Http404
    respuesta = FileResponse(almacenamiento.open(ruta, "rb"), as_attachment=True, filename=nombre)
    respuesta["X-Content-Type-Options"] = "nosniff"
    return respuesta

from decimal import ROUND_HALF_UP, Decimal, InvalidOperation

from django.conf import settings

from apps.core.constants import CENTAVO, TASA_IMPUESTO_DEFECTO


def tasa_impuesto():
    return Decimal(str(getattr(settings, "TASA_IMPUESTO", TASA_IMPUESTO_DEFECTO)))


def redondear(valor):
    return Decimal(valor).quantize(CENTAVO, rounding=ROUND_HALF_UP)


def subtotal_linea(cantidad, precio_unitario, descuento=Decimal("0")):
    return redondear(Decimal(cantidad) * Decimal(precio_unitario) - Decimal(descuento))


def calcular_linea(cantidad, precio_unitario, descuento=Decimal("0"), tasa=None):
    tasa = tasa_impuesto() if tasa is None else Decimal(str(tasa))
    descuento = redondear(descuento)
    subtotal = subtotal_linea(cantidad, precio_unitario, descuento)
    return {"subtotal": subtotal, "descuento": descuento, "impuesto": redondear(subtotal * tasa)}


def totales_documento(lineas):
    lineas = list(lineas)
    bruto = sum((linea["subtotal"] + linea["descuento"] for linea in lineas), Decimal("0"))
    descuento = sum((linea["descuento"] for linea in lineas), Decimal("0"))
    impuesto = sum((linea["impuesto"] for linea in lineas), Decimal("0"))
    return {
        "subtotal": redondear(bruto),
        "descuento": redondear(descuento),
        "impuesto": redondear(impuesto),
        "total": redondear(bruto - descuento + impuesto),
    }


def importes_coherentes(subtotal, descuento, impuesto, total):
    try:
        esperado = redondear(subtotal) - redondear(descuento) + redondear(impuesto)
        return esperado == redondear(total)
    except (InvalidOperation, TypeError, ValueError):
        return True


def linea_coherente(cantidad, precio_unitario, descuento, subtotal):
    try:

    except (InvalidOperation, TypeError, ValueError):

from django.db import models


class ModeloValidado(models.Model):
    campos_texto_opcional = ()

    class Meta:
        abstract = True

    def normalizar(self):
        for campo in self.campos_texto_opcional:
            valor = getattr(self, campo)
            if isinstance(valor, str):
                setattr(self, campo, valor.strip() or None)

    def clean(self):
        super().clean()
        self.normalizar()

    def save(self, *args, **kwargs):
        update_fields = kwargs.get("update_fields")
        if update_fields is None:
            self.full_clean()
        else:
            nombres = set(update_fields)
            excluir = [
                campo.name
                for campo in self._meta.concrete_fields
                if campo.name not in nombres and campo.attname not in nombres
            ]
            self.clean_fields(exclude=excluir)
        super().save(*args, **kwargs)

def sembrar_datos_iniciales(sender=None, using="default", **kwargs):
    from apps.catalogo.models import CategoriaServicio
    from apps.pagos.models import MetodoPago
    from apps.usuarios.constants import GRUPOS
    from apps.usuarios.models import EstadoUsuario

    conexion = connections[using]
    existentes = set(conexion.introspection.table_names())
    requeridas = {modelo._meta.db_table for modelo in (EstadoUsuario, CategoriaServicio, MetodoPago, Group)}
    if not requeridas <= existentes:
        return

    with transaction.atomic(using=using):
        for id_, nombre, descripcion in ESTADOS_USUARIO:
            EstadoUsuario.objects.using(using).get_or_create(
                nombre=nombre, defaults={"id": id_, "descripcion": descripcion}
            )

            MetodoPago.objects.using(using).get_or_create(

            )
        for nombre, descripcion in CATEGORIAS_SERVICIOS:
            CategoriaServicio.objects.using(using).get_or_create(
                nombre=nombre, defaults={"descripcion": descripcion}
            )
        for nombre in GRUPOS:
            Group.objects.using(using).get_or_create(name=nombre)

        sentencias = conexion.ops.sequence_reset_sql(no_style(), [EstadoUsuario, MetodoPago])
        with conexion.cursor() as cursor:
            for sentencia in sentencias:
                cursor.execute(sentencia)

def validar_ruta_relativa(valor):
    texto = str(valor)
    partes = re.split(r"[\\/]", texto)
    if (
        not texto
        or len(texto) > 500
        or texto != texto.strip()
        or texto.startswith(("/", "\\", "~"))
        or re.match(r"^[A-Za-z]:", texto)
        or "://" in texto
        or any(parte in ("", ".", "..") for parte in partes)
        or any(ord(caracter) < 32 for caracter in texto)
    ):
        raise ValidationError("Ruta de archivo no permitida.", code="ruta_invalida")


def validar_nombre_archivo(valor):
    texto = str(valor)
    if (
        not texto
        or texto != texto.strip()
        or texto.startswith(".")
        or re.search(r"[\\/\x00-\x1f]", texto)
    ):
        raise ValidationError("Nombre de archivo no permitido.", code="archivo_invalido")


def validar_tamano_archivo(valor):
    maximo = getattr(settings, "MAX_ARCHIVO_BYTES", MAX_ARCHIVO_BYTES)
    if valor > maximo:
        raise ValidationError(f"El archivo supera el maximo permitido de {maximo} bytes.", code="archivo_grande")


def validar_objeto_json(valor):
    if valor is None:
        return
    if not isinstance(valor, dict):
        raise ValidationError("Debe ser un objeto JSON.", code="json_invalido")
    try:
        longitud = len(json.dumps(valor, ensure_ascii=False))
    except (TypeError, ValueError):
        raise ValidationError("El objeto JSON no es serializable.", code="json_invalido")
    if longitud > MAX_JSON_BYTES:
        raise ValidationError("El objeto JSON es demasiado grande.", code="json_grande")

import uuid
from datetime import timedelta
from decimal import Decimal

from django.conf import settings
from django.core.exceptions import PermissionDenied

from apps.core.archivos import validar_archivo

from apps.cotizaciones.models import ArchivoCotizacion, Cotizacion, CotizacionDetalle, EstadoCotizacion

from apps.usuarios.constants import GRUPO_ADMINISTRADOR, GRUPO_OPERADOR

E = EstadoCotizacion

TRANSICIONES = {
    E.SOLICITADA.value: {E.BORRADOR.value, E.CANCELADA.value},
    E.BORRADOR.value: {E.ENVIADA.value, E.CANCELADA.value},
    E.ENVIADA.value: {E.ACEPTADA.value, E.RECHAZADA.value, E.VENCIDA.value, E.CANCELADA.value},
    E.ACEPTADA.value: set(),
    E.RECHAZADA.value: set(),
    E.VENCIDA.value: set(),
    E.CANCELADA.value: set(),
}
EDITABLES = (E.SOLICITADA.value, E.BORRADOR.value)


class ErrorCotizacion(Exception):
    pass


class TransicionInvalida(ErrorCotizacion):
    pass


class CotizacionVencida(ErrorCotizacion):
    pass


class CotizacionIncompleta(ErrorCotizacion):
    pass


def _es_personal(usuario):
    return usuario.tiene_algun_rol((GRUPO_OPERADOR, GRUPO_ADMINISTRADOR))


def _exigir_personal(usuario):
    if not _es_personal(usuario):
        raise PermissionDenied("Solo el personal autorizado puede realizar esta acción.")


def _exigir_propietario(cotizacion, usuario):
    if cotizacion.cliente.usuario_id != usuario.pk:
        raise PermissionDenied("Esta cotización pertenece a otro cliente.")


def _bloquear(cotizacion_id):
    return (
        Cotizacion.objects.select_for_update(of=("self",))
        .select_related("cliente")
        .get(pk=cotizacion_id)
    )


def _cambiar_estado(cotizacion, nuevo):
    nuevo = str(nuevo)
    if nuevo not in TRANSICIONES.get(cotizacion.estado, set()):
        raise TransicionInvalida(
            f"No se puede pasar de {cotizacion.get_estado_display()} a {EstadoCotizacion(nuevo).label}."
        )
    cotizacion.estado = nuevo


def _registrar(cotizacion, asunto, usuario=None):
    responsable = usuario if usuario is not None and _es_personal(usuario) else cotizacion.usuario_responsable
    if responsable is None:
        return
    InteraccionCliente.objects.create(
        cliente=cotizacion.cliente,
        usuario_responsable=responsable,
        tipo_interaccion=TipoInteraccion.SEGUIMIENTO,
        asunto=asunto[:150],
        descripcion=f"Cotización {cotizacion.codigo}",
    )


def _vigente(cotizacion):
    return not (cotizacion.fecha_vencimiento and cotizacion.fecha_vencimiento < timezone.now())


@transaction.atomic
def solicitar_cotizacion(cliente, usuario, servicio, cantidad, descripcion, solicitud, especificaciones=None, archivos=()):
    if not servicio.activo or not servicio.categoria.activo:
        raise ErrorCotizacion("El servicio seleccionado no está disponible.")
    maximo = getattr(settings, "COTIZACION_MAX_ARCHIVOS", 5)
    if len(archivos) > maximo:
        raise ErrorCotizacion(f"Puede adjuntar hasta {maximo} archivos.")
    validados = [(archivo, *validar_archivo(archivo)) for archivo in archivos]

    linea = calcular_linea(cantidad, servicio.precio_base, Decimal("0"))
    cotizacion = Cotizacion.objects.create(
        cliente=cliente,
        codigo=f"TMP-{uuid.uuid4().hex[:12].upper()}",
        estado=E.SOLICITADA,
        solicitud=solicitud,
        **totales_documento([linea]),
    )
    cotizacion.codigo = f"COT-{timezone.now():%Y}-{cotizacion.pk:06d}"
    cotizacion.save(update_fields=["codigo", "actualizado_en"])
    CotizacionDetalle.objects.create(
        cotizacion=cotizacion,
        servicio=servicio,
        descripcion=descripcion[:250],
        cantidad=cantidad,
        precio_unitario=servicio.precio_base,
        descuento=linea["descuento"],
        impuesto=linea["impuesto"],
        subtotal=linea["subtotal"],
        especificaciones=especificaciones or None,
    )

    guardados = []
    try:
        for archivo, formato, nombre in validados:
            ruta = almacenamiento_privado.save(
                f"cotizaciones/{cotizacion.pk}/{uuid.uuid4().hex}.{formato.lower()}", archivo
            )
            guardados.append(ruta)
            ArchivoCotizacion.objects.create(
                cotizacion=cotizacion,
                nombre_archivo=nombre,
                ruta_archivo=ruta,
                formato=formato,
                tamano_bytes=archivo.size,
                usuario_subida=usuario,
            )
    except Exception:
        for ruta in guardados:
            almacenamiento_privado.delete(ruta)
        raise
    return cotizacion


@transaction.atomic
def revisar_cotizacion(cotizacion_id, usuario, lineas, observaciones=None, plazo_entrega_dias=None):
    _exigir_personal(usuario)
    cotizacion = _bloquear(cotizacion_id)
    if cotizacion.estado not in EDITABLES:
        raise TransicionInvalida("Solo se pueden editar cotizaciones solicitadas o en borrador.")
    detalles = {d.pk: d for d in cotizacion.detalles.select_for_update()}
    ids = [datos["id"] for datos in lineas]
    if not ids or len(ids) != len(set(ids)) or set(ids) != set(detalles):
        raise ErrorCotizacion("Debe indicar todas las líneas de la cotización, sin repetir.")

    calculadas = []
    for datos in lineas:
        detalle = detalles[datos["id"]]
        calculo = calcular_linea(datos["cantidad"], datos["precio_unitario"], datos.get("descuento") or Decimal("0"))
        detalle.cantidad = datos["cantidad"]
        detalle.precio_unitario = datos["precio_unitario"]
        detalle.descuento = calculo["descuento"]
        detalle.impuesto = calculo["impuesto"]
        detalle.subtotal = calculo["subtotal"]
        detalle.save()
        calculadas.append(calculo)

    for campo, valor in totales_documento(calculadas).items():
        setattr(cotizacion, campo, valor)
    cotizacion.observaciones = observaciones
    cotizacion.plazo_entrega_dias = plazo_entrega_dias
    cotizacion.usuario_responsable = usuario
    if cotizacion.estado == E.SOLICITADA.value:
        _cambiar_estado(cotizacion, E.BORRADOR)
    cotizacion.save()
    _registrar(cotizacion, "Cotización revisada", usuario)
    return cotizacion


@transaction.atomic
def enviar_cotizacion(cotizacion_id, usuario):
    _exigir_personal(usuario)
    cotizacion = _bloquear(cotizacion_id)
    if cotizacion.total <= 0 or not cotizacion.detalles.exists():
        raise CotizacionIncompleta("La cotización debe tener importes definidos antes de enviarse.")
    _cambiar_estado(cotizacion, E.ENVIADA)
    cotizacion.usuario_responsable = cotizacion.usuario_responsable or usuario
    vigencia = getattr(settings, "COTIZACION_VIGENCIA_DIAS", 15)
    cotizacion.fecha_vencimiento = timezone.now() + timedelta(days=vigencia)
    cotizacion.save()
    _registrar(cotizacion, "Cotización enviada al cliente", usuario)
    return cotizacion


@transaction.atomic
def _responder(cotizacion_id, usuario, nuevo, asunto):
    cotizacion = _bloquear(cotizacion_id)
    _exigir_propietario(cotizacion, usuario)
    if cotizacion.estado == E.ENVIADA.value and not _vigente(cotizacion):
        raise CotizacionVencida("La cotización ya venció.")
    _cambiar_estado(cotizacion, nuevo)
    cotizacion.save()
    _registrar(cotizacion, asunto)
    return cotizacion


def aceptar_cotizacion(cotizacion_id, usuario):
    try:
        return _responder(cotizacion_id, usuario, E.ACEPTADA, "Cotización aceptada por el cliente")
    except CotizacionVencida:
        marcar_vencida(cotizacion_id)
        raise


def rechazar_cotizacion(cotizacion_id, usuario):

    except CotizacionVencida:

        raise


@transaction.atomic
def cancelar_cotizacion(cotizacion_id, usuario):
    cotizacion = _bloquear(cotizacion_id)
    if not _es_personal(usuario):
        _exigir_propietario(cotizacion, usuario)
        if cotizacion.estado != E.SOLICITADA.value:
            raise TransicionInvalida("Solo puede cancelar solicitudes que aún no fueron revisadas.")
    _cambiar_estado(cotizacion, E.CANCELADA)
    cotizacion.save()
    _registrar(cotizacion, "Cotización cancelada", usuario)
    return cotizacion


@transaction.atomic
def marcar_vencida(cotizacion_id):
    cotizacion = _bloquear(cotizacion_id)
    if cotizacion.estado == E.ENVIADA.value and not _vigente(cotizacion):
        _cambiar_estado(cotizacion, E.VENCIDA)
        cotizacion.save()
        _registrar(cotizacion, "Cotización vencida")
        return True
    return False


def vencer_cotizaciones_vencidas():
    ids = list(
        Cotizacion.objects.filter(estado=E.ENVIADA, fecha_vencimiento__lt=timezone.now()).values_list("pk", flat=True)
    )
    return sum(1 for pk in ids if marcar_vencida(pk))

from django.core.exceptions import ValidationError
from django.core.validators import MaxValueValidator, MinValueValidator
from django.db import models
from django.db.models import Q
from django.utils import timezone

from apps.core.constraints import restriccion
from apps.core.models.base import ModeloValidado


class Resena(ModeloValidado):
    cliente = models.ForeignKey(
        "clientes.Cliente",
        on_delete=models.PROTECT,
        db_column="id_cliente",
        related_name="resenas",
    )
    servicio = models.ForeignKey(
        "catalogo.Servicio",
        on_delete=models.PROTECT,
        null=True,
        blank=True,
        db_column="id_servicio",
        related_name="resenas",
    )
    encargo = models.ForeignKey(
        "encargos.Encargo",
        on_delete=models.PROTECT,
        null=True,
        blank=True,
        db_column="id_encargo",
        related_name="resenas",
    )
    puntuacion = models.SmallIntegerField(validators=[MinValueValidator(1), MaxValueValidator(5)])
    comentario = models.TextField(null=True, blank=True)
    fecha_creacion = models.DateTimeField(default=timezone.now)

    campos_texto_opcional = ("comentario",)

    class Meta:
        db_table = "resenas"
        verbose_name = "reseña"
        verbose_name_plural = "reseñas"
        constraints = [
            restriccion("ck_resenas_puntuacion", Q(puntuacion__gte=1, puntuacion__lte=5)),
            restriccion(
                "ck_resenas_referencia",
                Q(servicio__isnull=False) | Q(encargo__isnull=False),
            ),
        ]

    def __str__(self):
        return f"Reseña {self.pk} ({self.puntuacion}/5)"

    def clean(self):
        super().clean()
        if not self.servicio_id and not self.encargo_id:
            raise ValidationError("La reseña debe referirse a un servicio o a un encargo.")
        if self.encargo_id and self.encargo.cliente_id != self.cliente_id:
            raise ValidationError({"encargo": "El encargo pertenece a otro cliente."})

                "ck_fact_det_importes",
                no_negativos("precio_unitario", "descuento", "impuesto", "subtotal"),
            ),
            restriccion(
                "ck_fact_det_tipo_referencia",
                Q(servicio__isnull=True) | Q(producto__isnull=True),
            ),
        ]

    def __str__(self):
        return f"{self.factura_id} - {self.descripcion}"

    def clean(self):
        super().clean()
        if self.servicio_id and self.producto_id:
            raise ValidationError("Un detalle no puede referirse a un servicio y a un producto a la vez.")

class EstadoPago(models.TextChoices):
    PENDIENTE = "PENDIENTE", "Pendiente"
    CONFIRMADO = "CONFIRMADO", "Confirmado"
    RECHAZADO = "RECHAZADO", "Rechazado"
    REEMBOLSADO = "REEMBOLSADO", "Reembolsado"
    CANCELADO = "CANCELADO", "Cancelado"


class Pago(ModeloValidado):
    factura = models.ForeignKey(
        "facturacion.Factura",
        on_delete=models.PROTECT,
        db_column="id_factura",
        related_name="pagos",
    )
    metodo_pago = models.ForeignKey(
        "pagos.MetodoPago",
        on_delete=models.PROTECT,
        db_column="id_metodo_pago",
        related_name="pagos",
    )
    referencia = models.CharField(max_length=120, null=True, blank=True)
    monto = models.DecimalField(max_digits=12, decimal_places=2, validators=[MinValueValidator(Decimal("0.01"))])
    fecha_pago = models.DateTimeField(default=timezone.now)
    estado = models.CharField(max_length=20, choices=EstadoPago.choices, default=EstadoPago.PENDIENTE)
    registrado_por = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        db_column="registrado_por",
        related_name="pagos_registrados",
    )
    observaciones = models.TextField(null=True, blank=True)

    campos_texto_opcional = ("referencia", "observaciones")

    class Meta:
        db_table = "pagos"
        verbose_name = "pago"
        verbose_name_plural = "pagos"
        constraints = [
            restriccion("ck_pagos_monto", Q(monto__gt=0)),
            en_lista("ck_pagos_estado", "estado", EstadoPago.values),
        ]
        indexes = [
            models.Index(fields=["estado"], name="idx_pagos_estado"),
            models.Index(fields=["fecha_pago"], name="idx_pagos_fecha"),
        ]

    def __str__(self):
        return f"Pago {self.pk} - {self.monto}"

    def clean(self):
        super().clean()
        if self._state.adding and self.factura_id:
            factura = self.factura
            if factura.estado == EstadoFactura.ANULADA:
                raise ValidationError({"factura": "No se pueden registrar pagos sobre una factura anulada."})
            if self.monto is not None and self.monto > factura.total:
                raise ValidationError({"monto": "El monto no puede superar el total de la factura."})

    context_object_name = "productos"
    paginate_by = 20

            solo_bajo_stock=bool(self.request.GET.get("bajo")),
        )

    def get_context_data(self, **kwargs):
        contexto = super().get_context_data(**kwargs)
        contexto["q"] = self.request.GET.get("q", "").strip()
        contexto["bajo"] = bool(self.request.GET.get("bajo"))
        return contexto

    form_class = ProductoForm
    titulo = "Nuevo producto"
    mensaje_exito = "Producto creado correctamente."
    success_url = reverse_lazy("administrator:inventario")


class ProductoEditarView(FormularioAdminMixin, UpdateView):
    model = Producto
    form_class = ProductoForm
    titulo = "Editar producto"
    mensaje_exito = "Producto actualizado correctamente."
    success_url = reverse_lazy("administrator:inventario")


class ProductoActivoView(AdministradorRequeridoMixin, View):
    http_method_names = ["post"]

    def post(self, request, pk):
        producto = get_object_or_404(Producto, pk=pk)
        producto.activo = not producto.activo
        producto.save(update_fields=["activo", "actualizado_en"])
        messages.success(request, "Producto activado." if producto.activo else "Producto desactivado.")
        return redirect("administrator:inventario")


class ProductoStockView(AdministradorRequeridoMixin, View):
    http_method_names = ["post"]

    def post(self, request, pk):
        form = AjusteStockForm(request.POST)
        if not form.is_valid():
            messages.error(request, "Ingrese un ajuste de stock válido.")
            return redirect("administrator:inventario")
        try:
            producto = servicio_catalogo.ajustar_stock(pk, form.cleaned_data["delta"])
        except Producto.DoesNotExist:
            raise Http404
        except servicio_catalogo.ErrorInventario as error:
            messages.error(request, str(error))
        else:
            messages.success(request, f"Stock de {producto.nombre}: {producto.stock}.")
        return redirect("administrator:inventario")

from django.http import Http404

from apps.clientes.services.clientes import asegurar_cliente

from apps.cotizaciones.selectors import cotizaciones as selectors
from apps.cotizaciones.services import cotizaciones as servicios
from apps.portal.mixins import ClienteRequeridoMixin

ACCIONES = {
    "aceptar": ("aceptar_cotizacion", "Cotización aceptada. Gracias por su confianza."),
    "rechazar": ("rechazar_cotizacion", "Cotización rechazada."),
    "cancelar": ("cancelar_cotizacion", "Solicitud cancelada."),
}


class CotizacionesView(ClienteRequeridoMixin, ListView):
    template_name = "customer/cotizaciones.html"
    context_object_name = "cotizaciones"
    paginate_by = 10

    def get_queryset(self):
        return selectors.cotizaciones_del_cliente(asegurar_cliente(self.request.user))


class CotizacionDetalleView(ClienteRequeridoMixin, View):
    template_name = "customer/cotizacion_detalle.html"

    def get(self, request, pk):
        cotizacion = selectors.obtener_del_cliente(asegurar_cliente(request.user), pk)
        if cotizacion is None:
            raise Http404
        return render(request, self.template_name, {"cotizacion": cotizacion})


class CotizacionAccionView(ClienteRequeridoMixin, View):
    http_method_names = ["post"]

    def post(self, request, pk, accion):
        if accion not in ACCIONES:
            raise Http404
        cotizacion = selectors.obtener_del_cliente(asegurar_cliente(request.user), pk)
        if cotizacion is None:
            raise Http404
        nombre_funcion, mensaje = ACCIONES[accion]
        try:
            getattr(servicios, nombre_funcion)(cotizacion.pk, request.user)
        except servicios.ErrorCotizacion as error:
            messages.error(request, str(error))
        else:
            messages.success(request, mensaje)
        return redirect("customer:cotizacion", pk=pk)


class CotizacionArchivoView(ClienteRequeridoMixin, View):
    def get(self, request, pk, archivo_id):
        cotizacion = selectors.obtener_del_cliente(asegurar_cliente(request.user), pk)
        if cotizacion is None:
            raise Http404
        archivo = get_object_or_404(cotizacion.archivos.all(), pk=archivo_id)
        return respuesta_descarga(almacenamiento_privado, archivo.ruta_archivo, archivo.nombre_archivo)

from django.contrib import messages
from django.core.exceptions import ValidationError
from django.shortcuts import redirect, render
from django.views import View

from apps.clientes.services.clientes import asegurar_cliente
from apps.core.utiles import mensaje_de
from apps.cotizaciones.forms.cotizaciones import SolicitudCotizacionForm
from apps.cotizaciones.services import cotizaciones as servicios
from apps.portal.mixins import ClienteRequeridoMixin


class CotizarView(ClienteRequeridoMixin, View):
    template_name = "customer/cotizar.html"

    def get(self, request):
        return render(request, self.template_name, {"form": SolicitudCotizacionForm()})

    def post(self, request):
        form = SolicitudCotizacionForm(request.POST, request.FILES)
        if form.is_valid():
            try:
                cotizacion = servicios.solicitar_cotizacion(
                    cliente=asegurar_cliente(request.user),
                    usuario=request.user,
                    servicio=form.cleaned_data["servicio"],
                    cantidad=form.cleaned_data["cantidad"],
                    descripcion=form.cleaned_data["descripcion"],
                    solicitud=form.cleaned_data["solicitud"],
                    especificaciones=form.especificaciones(),
                    archivos=form.cleaned_data["archivos"],
                )
            except (servicios.ErrorCotizacion, ValidationError) as error:
                form.add_error(None, mensaje_de(error))
            else:
                messages.success(request, f"Su solicitud {cotizacion.codigo} fue recibida.")
                return redirect("customer:cotizacion", pk=cotizacion.pk)
        return render(request, self.template_name, {"form": form})

from django.contrib import messages
from django.shortcuts import redirect, render
from django.views import View

from apps.clientes.forms.clientes import PerfilClienteForm
from apps.clientes.selectors import clientes as selectors
from apps.clientes.services import clientes as servicios
from apps.portal.mixins import ClienteRequeridoMixin


class CuentaView(ClienteRequeridoMixin, View):
    template_name = "customer/cuenta.html"

    def _contexto(self, form, cliente):
        return {"form": form, "cliente": cliente, "actividad": selectors.actividad_reciente(cliente)}

    def get(self, request):
        cliente = servicios.asegurar_cliente(request.user)
        return render(request, self.template_name, self._contexto(PerfilClienteForm(instance=cliente), cliente))

    def post(self, request):
        cliente = servicios.asegurar_cliente(request.user)
        form = PerfilClienteForm(request.POST, instance=cliente)
        if form.is_valid():
            servicios.guardar_perfil(form)
            messages.success(request, "Sus datos se actualizaron correctamente.")
            return redirect("customer:cuenta")
        return render(request, self.template_name, self._contexto(form, cliente))

from django.shortcuts import redirect
from django.utils.http import url_has_allowed_host_and_scheme
from django.views.generic import TemplateView


class AccesoView(TemplateView):
    template_name = "guest/acceso.html"

    def dispatch(self, request, *args, **kwargs):
        if request.user.is_authenticated:
            return redirect("redirigir")
        return super().dispatch(request, *args, **kwargs)

    def get_context_data(self, **kwargs):
        contexto = super().get_context_data(**kwargs)
        siguiente = self.request.GET.get("next", "")
        if not url_has_allowed_host_and_scheme(siguiente, {self.request.get_host()}, self.request.is_secure()):
            siguiente = ""
        contexto["siguiente"] = siguiente
        return contexto

from django.contrib import messages

from django.http import Http404

from apps.cotizaciones.services import cotizaciones as servicios
from apps.portal.mixins import OperadorRequeridoMixin

ACCIONES = {
    "enviar": ("enviar_cotizacion", "Cotización enviada al cliente."),
    "cancelar": ("cancelar_cotizacion", "Cotización cancelada."),
}


class CotizacionesView(OperadorRequeridoMixin, ListView):
    template_name = "operator/cotizaciones.html"
    context_object_name = "cotizaciones"
    paginate_by = 20

    def get_queryset(self):
        return selectors.bandeja_operador(
            self.request.GET.get("estado", ""),
            self.request.GET.get("q", "").strip(),
        )

    def get_context_data(self, **kwargs):

        contexto["estados"] = EstadoCotizacion.choices
        contexto["estado"] = self.request.GET.get("estado", "")
        contexto["q"] = self.request.GET.get("q", "").strip()
        contexto["conteos"] = selectors.contar_por_estado()
        return contexto


class CotizacionOperadorView(OperadorRequeridoMixin, View):
    template_name = "operator/cotizacion.html"

    def _cotizacion(self, pk):
        cotizacion = selectors.obtener_para_operador(pk)
        if cotizacion is None:
            raise Http404
        return cotizacion

    def _iniciales(self, cotizacion):
        return [
            {
                "id": detalle.pk,
                "descripcion": detalle.descripcion,
                "cantidad": detalle.cantidad,
                "precio_unitario": detalle.precio_unitario,
                "descuento": detalle.descuento,
            }
            for detalle in cotizacion.detalles.all()
        ]

    def _responder(self, request, cotizacion, form=None, formset=None):
        editable = cotizacion.estado in servicios.EDITABLES
        if editable and form is None:
            form = RevisionCotizacionForm(
                initial={
                    "observaciones": cotizacion.observaciones,
                    "plazo_entrega_dias": cotizacion.plazo_entrega_dias,
                }
            )
            formset = LineasFormSet(initial=self._iniciales(cotizacion), prefix="lineas")
        contexto = {"cotizacion": cotizacion, "editable": editable, "form": form, "formset": formset}
        return render(request, self.template_name, contexto)

    def get(self, request, pk):
        return self._responder(request, self._cotizacion(pk))

    def post(self, request, pk):
        cotizacion = self._cotizacion(pk)
        if cotizacion.estado not in servicios.EDITABLES:
            messages.error(request, "Esta cotización ya no se puede editar.")
            return redirect("operator:cotizacion", pk=pk)
        form = RevisionCotizacionForm(request.POST)
        formset = LineasFormSet(request.POST, initial=self._iniciales(cotizacion), prefix="lineas")
        if form.is_valid() and formset.is_valid():
            try:
                servicios.revisar_cotizacion(
                    cotizacion.pk,
                    request.user,
                    lineas=formset.cleaned_data,
                    observaciones=form.cleaned_data["observaciones"] or None,
                    plazo_entrega_dias=form.cleaned_data["plazo_entrega_dias"],
                )
            except (servicios.ErrorCotizacion, ValidationError) as error:
                messages.error(request, mensaje_de(error))
            else:
                messages.success(request, "Cotización actualizada.")
                return redirect("operator:cotizacion", pk=pk)
        return self._responder(request, cotizacion, form, formset)


class CotizacionOperadorAccionView(OperadorRequeridoMixin, View):
    http_method_names = ["post"]

    def post(self, request, pk, accion):
        if accion not in ACCIONES:
            raise Http404
        if selectors.obtener_para_operador(pk) is None:
            raise Http404
        nombre_funcion, mensaje = ACCIONES[accion]
        try:
            getattr(servicios, nombre_funcion)(pk, request.user)
        except (servicios.ErrorCotizacion, ValidationError) as error:
            messages.error(request, mensaje_de(error))
        else:
            messages.success(request, mensaje)
        return redirect("operator:cotizacion", pk=pk)


class CotizacionOperadorArchivoView(OperadorRequeridoMixin, View):
    def get(self, request, pk, archivo_id):
        cotizacion = selectors.obtener_para_operador(pk)
        if cotizacion is None:
            raise Http404
        archivo = get_object_or_404(cotizacion.archivos.all(), pk=archivo_id)
        return respuesta_descarga(almacenamiento_privado, archivo.ruta_archivo, archivo.nombre_archivo)

from allauth.account.adapter import DefaultAccountAdapter
from allauth.socialaccount.adapter import DefaultSocialAccountAdapter
from django.core.exceptions import ValidationError

from apps.core.validators import validar_nombre_persona
from apps.usuarios.constants import ESTADO_ACTIVO, ESTADO_PENDIENTE


def _nombre_o_defecto(valor, defecto):
    valor = (valor or "").strip()
    try:
        validar_nombre_persona(valor)
    except ValidationError:
        return defecto
    return valor


class CuentaAdapter(DefaultAccountAdapter):
    def is_open_for_signup(self, request):
        return True

    def save_user(self, request, user, form, commit=True):
        datos = form.cleaned_data
        correo = datos.get("email")
        if correo:
            user.correo = correo.strip().lower()
        if datos.get("nombres"):
            user.nombres = datos["nombres"].strip()
        if datos.get("apellidos"):
            user.apellidos = datos["apellidos"].strip()
        if "telefono" in datos:
            user.telefono = datos["telefono"] or None
        user.is_staff = False
        user.is_superuser = False
        if getattr(form, "sociallogin", None) is None:
            user.estado_id = ESTADO_PENDIENTE
        password = datos.get("password1")
        if password:
            user.set_password(password)
        else:
            user.set_unusable_password()
        if commit:
            user.save()
        return user


class SocialAdapter(DefaultSocialAccountAdapter):
    def is_open_for_signup(self, request, sociallogin):
        return True

    def populate_user(self, request, sociallogin, data):
        usuario = sociallogin.user
        usuario.correo = (data.get("email") or "").strip().lower()
        usuario.nombres = _nombre_o_defecto(data.get("first_name"), "Cliente")
        usuario.apellidos = _nombre_o_defecto(data.get("last_name"), "Sin apellido")
        usuario.is_staff = False
        usuario.is_superuser = False
        usuario.estado_id = ESTADO_ACTIVO
        usuario.set_unusable_password()
        return usuario

from django.contrib.auth.models import BaseUserManager

from apps.usuarios.constants import ESTADO_ACTIVO


class UsuarioManager(BaseUserManager):
    def get_by_natural_key(self, correo):
        return self.get(**{f"{self.model.USERNAME_FIELD}__iexact": correo})

    def _crear(self, correo, password, **extra):
        if not correo:
            raise ValueError("El correo es obligatorio.")
        extra.setdefault("estado_id", ESTADO_ACTIVO)
        usuario = self.model(correo=self.normalize_email(correo).lower(), **extra)
        if password:
            usuario.set_password(password)
        else:
            usuario.set_unusable_password()
        usuario.save(using=self._db)
        return usuario

    def create_user(self, correo, password=None, **extra):
        extra.setdefault("is_staff", False)
        extra.setdefault("is_superuser", False)
        return self._crear(correo, password, **extra)

    def create_superuser(self, correo, password=None, **extra):
        extra.setdefault("is_staff", True)
        extra.setdefault("is_superuser", True)
        if extra.get("is_staff") is not True or extra.get("is_superuser") is not True:
            raise ValueError("El superusuario debe tener is_staff e is_superuser en True.")
        return self._crear(correo, password, **extra)

from django.contrib.auth.models import AbstractBaseUser, PermissionsMixin
from django.db import models
from django.utils import timezone

from apps.core.models.base import ModeloValidado
from apps.core.validators import validar_nombre_persona, validar_telefono
from apps.usuarios.constants import ESTADO_ACTIVO, ESTADO_INACTIVO, ESTADOS_BLOQUEADOS
from apps.usuarios.managers import UsuarioManager


class Usuario(ModeloValidado, AbstractBaseUser, PermissionsMixin):
    nombres = models.CharField(max_length=100, validators=[validar_nombre_persona])
    apellidos = models.CharField(max_length=100, validators=[validar_nombre_persona])
    correo = models.EmailField(max_length=150, unique=True)
    telefono = models.CharField(max_length=25, null=True, blank=True, validators=[validar_telefono])
    is_staff = models.BooleanField(default=False)
    is_active = models.BooleanField(default=True)
    date_joined = models.DateTimeField(default=timezone.now)
    estado = models.ForeignKey(
        "usuarios.EstadoUsuario",
        on_delete=models.PROTECT,
        db_column="id_estado",
        related_name="usuarios",
        default=ESTADO_ACTIVO,
    )

    objects = UsuarioManager()

    USERNAME_FIELD = "correo"
    EMAIL_FIELD = "correo"
    REQUIRED_FIELDS = ["nombres", "apellidos"]
    campos_texto_opcional = ("telefono",)

    class Meta:
        db_table = "usuarios"
        verbose_name = "usuario"
        verbose_name_plural = "usuarios"

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self._estado_inicial = self.__dict__.get("estado_id")

    def __str__(self):
        return self.correo

    @property
    def nombre_completo(self):
        return f"{self.nombres} {self.apellidos}".strip()

    def get_full_name(self):
        return self.nombre_completo

    def get_short_name(self):
        return self.nombres

    def clean(self):
        super().clean()
        if self.correo:
            self.correo = self.correo.strip().lower()
        if self.estado_id in ESTADOS_BLOQUEADOS:
            self.is_active = False
        elif self.estado_id == ESTADO_ACTIVO:
            if self._estado_inicial in ESTADOS_BLOQUEADOS:
                self.is_active = True
            elif not self.is_active:
                self.estado_id = ESTADO_INACTIVO

    def save(self, *args, **kwargs):
        super().save(*args, **kwargs)
        self._estado_inicial = self.estado_id

    def tiene_algun_rol(self, roles):
        if not (self.is_active and self.estado_id == ESTADO_ACTIVO):
            return False
        if self.is_superuser:
            return True
        return self.groups.filter(name__in=roles).exists()

def obtener_identificador(request, credentials=None):
    credentials = credentials or {}
    for clave in ("correo", "email", "login", "username"):
        valor = credentials.get(clave)
        if not valor and request is not None and hasattr(request, "POST"):
            valor = request.POST.get(clave)
        if valor:
            return str(valor).strip().lower()
    return None

@receiver(user_signed_up, dispatch_uid="usuarios.registrar_cliente")
def registrar_cliente(sender, request, user, **kwargs):
    if user.is_staff or user.is_superuser:
        return
    grupo, _ = Group.objects.get_or_create(name=GRUPO_CLIENTE)
    user.groups.add(grupo)
    asegurar_cliente(user)

    if usuario.estado_id == ESTADO_PENDIENTE:
        usuario.estado_id = ESTADO_ACTIVO
        usuario.save(update_fields=["estado"])

from django.urls import reverse

from apps.usuarios.constants import (

    GRUPO_ADMINISTRADOR,
    GRUPO_CLIENTE,
    GRUPO_OPERADOR,

from apps.usuarios.tests.utiles import crear_usuario, iniciar_sesion


class PermisosPorRolTests(TestCase):
    def setUp(self):
        self.cliente_u = crear_usuario("cliente@example.com", GRUPO_CLIENTE)
        self.operador_u = crear_usuario("operador@example.com", GRUPO_OPERADOR, is_staff=True)
        self.admin_u = crear_usuario("admin@example.com", GRUPO_ADMINISTRADOR, is_staff=True)

    def _codigo(self, usuario, nombre):
        if usuario is not None:
            iniciar_sesion(self.client, usuario)
        return self.client.get(reverse(nombre)).status_code

    def test_anonimo_es_redirigido_al_acceso(self):
        respuesta = self.client.get(reverse("customer:panel"))
        self.assertEqual(respuesta.status_code, 302)
        self.assertIn(reverse("guest:acceso"), respuesta.url)

    def test_cliente_solo_accede_a_su_area(self):
        self.assertEqual(self._codigo(self.cliente_u, "customer:panel"), 200)
        self.assertEqual(self._codigo(None, "operator:panel"), 403)
        self.assertEqual(self._codigo(None, "administrator:dashboard"), 403)
        self.assertEqual(self._codigo(None, "administrator:inventario"), 403)

    def test_operador_no_ejecuta_funciones_de_administrador(self):
        self.assertEqual(self._codigo(self.operador_u, "operator:panel"), 200)
        self.assertEqual(self._codigo(None, "administrator:dashboard"), 403)
        self.assertEqual(self._codigo(None, "administrator:producto_crear"), 403)
        self.assertEqual(self._codigo(None, "customer:panel"), 403)

    def test_administrador_accede_a_su_area(self):
        self.assertEqual(self._codigo(self.admin_u, "administrator:dashboard"), 200)
        self.assertEqual(self._codigo(None, "administrator:inventario"), 200)
        self.assertEqual(self._codigo(None, "administrator:servicios"), 200)

    def test_redireccion_posterior_al_login_segun_rol(self):
        esperados = {
            self.cliente_u: reverse("customer:panel"),
            self.operador_u: reverse("operator:panel"),
            self.admin_u: reverse("administrator:dashboard"),
        }
        for usuario, destino in esperados.items():
            self.client.logout()
            iniciar_sesion(self.client, usuario)
            self.assertRedirects(
                self.client.get(reverse("redirigir")), destino, fetch_redirect_response=False
            )

#!/usr/bin/env bash

readonly NOMBRE_SALIDA="project-context.txt"

# Tamaño máximo por archivo: 1 MB por defecto.
# Se puede modificar mediante la variable MAX_FILE_SIZE.
MAX_FILE_SIZE="${MAX_FILE_SIZE:-1048576}"

# Directorios que normalmente contienen dependencias,
# archivos temporales, cachés o resultados de compilación.
readonly -a EXCLUDE_DIRS=(
    ".git"
    ".svn"
    ".hg"
    "node_modules"
    "vendor"
    ".venv"
    "venv"
    "__pycache__"
    ".pytest_cache"
    ".mypy_cache"
    ".ruff_cache"
    ".tox"
    ".cache"
    ".idea"
    ".vscode"
    ".angular"
    ".next"
    ".nuxt"
    ".svelte-kit"
    ".turbo"
    ".gradle"
    ".terraform"
    ".dart_tool"
    "Pods"
    "DerivedData"
    "coverage"
    "dist"
    "build"
    "target"
    "out"
    "bin"
    "obj"
)

# Patrones de archivos que pueden contener información sensible.
readonly -a SENSITIVE_PATTERNS=(
    ".env"
    ".env.*"
    "*.pem"
    "*.key"

    "id_ed25519*"

    ".pypirc"

    "*secrets*.json"

)

    cat <<'EOF'
INSPECTOR UNIVERSAL DE PROYECTOS

Uso:
  ./inspeccionar-proyecto.sh [DIRECTORIO] [ARCHIVO_SALIDA]

Ejemplos:
  ./inspeccionar-proyecto.sh
  ./inspeccionar-proyecto.sh ./backend
  ./inspeccionar-proyecto.sh ./backend informe.txt
  ./inspeccionar-proyecto.sh ~/Projects/mi-proyecto ~/Documents/informe.txt

Variables de entorno:
  MAX_FILE_SIZE=2097152

La variable MAX_FILE_SIZE permite modificar el tamaño máximo
de cada archivo que se incorporará al informe (en bytes).

EOF
}

if [[ "${1:-}" == "-h" || "${1:-}" == "--help" ]]; then
    mostrar_ayuda
    exit 0
fi

if (( $# > 2 )); then
    echo "[ERROR] Número de argumentos incorrecto."
    mostrar_ayuda
    exit 1
fi

if [[ ! "$MAX_FILE_SIZE" =~ ^[0-9]+$ ]] || (( MAX_FILE_SIZE < 1 )); then
    echo "[ERROR] MAX_FILE_SIZE debe ser un entero positivo."
    exit 1
fi

DIRECTORIO="${1:-.}"
SALIDA="${2:-$NOMBRE_SALIDA}"

if [[ ! -d "$DIRECTORIO" ]]; then

fi

ROOT="$(cd -- "$DIRECTORIO" && pwd -P)"

# Resolver la ruta de salida.
if [[ "$SALIDA" != /* ]]; then
    SALIDA="$ROOT/$SALIDA"
fi

DIRECTORIO_SALIDA="$(dirname -- "$SALIDA")"
NOMBRE_ARCHIVO="$(basename -- "$SALIDA")"

    echo "[ERROR] La ruta de salida no puede ser un directorio."
    exit 1

if [[ -d "$SALIDA" ]]; then
    echo "[ERROR] La ruta de salida corresponde a un directorio."
    exit 1
fi

# ---------------------- FUNCIONES ---------------------------

es_archivo_sensible() {
    local nombre
    nombre="$(basename -- "$1")"

    for patron in "${SENSITIVE_PATTERNS[@]}"; do
        if [[ "$nombre" == $patron ]]; then
            return 0
        fi
    done

    return 1
}

buscar_elementos() {
    local -a argumentos

    argumentos=("$ROOT" "(" -type d "(")

    for directorio in "${EXCLUDE_DIRS[@]}"; do
        argumentos+=(-name "$directorio" -o)
    done

    # -false evita dejar un -o al final de la expresión.
    argumentos+=(
        -false
        ")"
        -prune
        ")"
        -o
        "("
        -type f
        -o
        -type d
        ")"
        -print0
    )

    find "${argumentos[@]}"
}

obtener_ruta_relativa() {
    local ruta="$1"

    if [[ "$ROOT" == "/" ]]; then
        printf '%s' "${ruta#/}"
    else
        printf '%s' "${ruta#"$ROOT"/}"
    fi
}

es_archivo_texto() {
    local archivo="$1"

    # Los archivos vacíos también son válidos.
    [[ ! -s "$archivo" ]] && return 0

    # grep -Iq permite detectar contenido binario habitual.
    grep -Iq . "$archivo"
}

echo "             INSPECTOR UNIVERSAL DE PROYECTOS"
echo "============================================================"
echo
echo "Directorio raíz : $ROOT"
echo "Archivo salida  : $SALIDA"
echo "Límite por archivo: $MAX_FILE_SIZE bytes"
echo

if [[ -e "$SALIDA" && ! -w "$SALIDA" ]]; then
    echo "[ERROR] No tienes permisos para sobrescribir el informe."

fi

echo "[INFO] Explorando directorios y archivos..."

mapfile -d '' -t ELEMENTOS < <(

TOTAL_ARCHIVOS=0
ARCHIVOS_INCLUIDOS=0
ARCHIVOS_OMITIDOS=0

    echo "              INFORME DEL PROYECTO"
    echo "============================================================"
    echo
    echo "Directorio raíz: $ROOT"
    echo "Fecha: $(date '+%Y-%m-%d %H:%M:%S')"
    echo "Límite por archivo: $MAX_FILE_SIZE bytes"
    echo

    # ========================================================
    # 1. ESTRUCTURA DEL PROYECTO
    # ========================================================

    echo "============================================================"
    echo " 1. ESTRUCTURA DE DIRECTORIOS Y ARCHIVOS"
    echo "============================================================"
    echo

    printf '[ROOT] %s/\n' "$(basename -- "$ROOT")"

    for elemento in "${ELEMENTOS[@]}"; do

        [[ "$elemento" == "$ROOT" ]] && continue
        [[ "$elemento" == "$SALIDA" ]] && continue

        relativo="$(obtener_ruta_relativa "$elemento")"

        separadores="${relativo//[^\/]/}"
        nivel="${#separadores}"

        if [[ -d "$elemento" ]]; then
            printf '%*s[DIR]  %s/\n' \
                "$((nivel * 4))" "" \
                "$(basename -- "$elemento")"

        elif [[ -f "$elemento" ]]; then
            printf '%*s[FILE] %s\n' \
                "$((nivel * 4))" "" \
                "$(basename -- "$elemento")"
        fi
    done

    echo

    # ========================================================
    # 2. CONTENIDO DE LOS ARCHIVOS
    # ========================================================

    echo "============================================================"
    echo " 2. CONTENIDO DE LOS ARCHIVOS"
    echo "============================================================"
    echo

    for elemento in "${ELEMENTOS[@]}"; do

        [[ ! -f "$elemento" ]] && continue
        [[ "$elemento" == "$SALIDA" ]] && continue

        ((TOTAL_ARCHIVOS += 1))

        relativo="$(obtener_ruta_relativa "$elemento")"

        echo "============================================================"
        echo "ARCHIVO: $relativo"
        echo "============================================================"
        echo

        # Comprobar información sensible.
        if es_archivo_sensible "$elemento"; then
            echo "[ARCHIVO OMITIDO: posible información sensible]"
            ((ARCHIVOS_OMITIDOS += 1))
            echo
            continue
        fi

        # Comprobar tamaño.
        tamano="$(wc -c < "$elemento")"

        if (( tamano > MAX_FILE_SIZE )); then
            echo "[ARCHIVO OMITIDO: supera el límite permitido]"
            echo "Tamaño: $tamano bytes"
            ((ARCHIVOS_OMITIDOS += 1))
            echo
            continue
        fi

        # Comprobar contenido binario.
        if ! es_archivo_texto "$elemento"; then
            echo "[ARCHIVO OMITIDO: contenido binario]"
            ((ARCHIVOS_OMITIDOS += 1))
            echo
            continue
        fi

        # Incorporar contenido.
        if [[ -s "$elemento" ]]; then
            cat -- "$elemento"
        else
            echo "[Archivo vacío]"
        fi

        ((ARCHIVOS_INCLUIDOS += 1))

        echo
        echo
    done

    # ========================================================
    # 3. RESUMEN
    # ========================================================

    echo "============================================================"
    echo " 3. RESUMEN DE LA INSPECCIÓN"
    echo "============================================================"
    echo
    echo "Archivos encontrados: $TOTAL_ARCHIVOS"
    echo "Archivos incluidos:   $ARCHIVOS_INCLUIDOS"
    echo "Archivos omitidos:    $ARCHIVOS_OMITIDOS"
    echo

} > "$SALIDA"

# ---------------------- FINALIZACIÓN ------------------------

echo "[OK] Inspección completada."
echo
echo "Archivos encontrados: $TOTAL_ARCHIVOS"
echo "Archivos incluidos:   $ARCHIVOS_INCLUIDOS"
echo "Archivos omitidos:    $ARCHIVOS_OMITIDOS"
echo
echo "Informe generado:"
echo "$SALIDA"
echo
du -h -- "$SALIDA"
echo
echo "============================================================"

    """Run administrative tasks."""
    os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'config.settings.development')
    try:
        from django.core.management import execute_from_command_line
    except ImportError as exc:
        raise ImportError(
            "Couldn't import Django. Are you sure it's installed and "
            "available on your PYTHONPATH environment variable? Did you "
            "forget to activate a virtual environment?"
        ) from exc
    execute_from_command_line(sys.argv)


if __name__ == '__main__':
    main()
